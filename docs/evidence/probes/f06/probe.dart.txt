// F06 probe (throwaway, not committed; kept as
// docs/evidence/probes/f06/probe.dart.txt): what Flutter exposes, and what
// screen readers say, for a text field with an error. One screen per
// fixture, opened by route: /p01 ... /p13. Every fixture's text starts with
// its id.
import 'package:flutter/cupertino.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';

import 'catalog/test_id.dart';

void main() {
  runApp(const ProbeApp());
  if (kIsWeb) SemanticsBinding.instance.ensureSemantics();
}

const _emailError = 'Enter an email address, like name@example.com.';
const _nameError = 'Enter your name.';
const _phoneError = 'Enter a phone number using digits only.';

String? _required(String? v, String error) =>
    (v == null || v.trim().isEmpty) ? error : null;

String? _email(String? v) =>
    (v == null || !RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(v))
    ? _emailError
    : null;

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'F06 probe',
      onGenerateRoute: (settings) {
        final name = settings.name ?? '/';
        final builder = _fixtures[name];
        return MaterialPageRoute<void>(
          settings: settings,
          builder: (_) => builder == null
              ? const _Index()
              : _Screen(title: '${name.substring(1)} fixture', child: builder()),
        );
      },
    );
  }
}

final Map<String, Widget Function()> _fixtures = {
  '/p01': () => const _StaticError(merged: true),
  '/p02': () => const _StaticError(merged: false),
  '/p03': () => const _ButtonSetsError(),
  '/p04': () => const _FormOne(),
  '/p05': () => const _FormTwo(id: 'p05', focus: _Focus.none),
  '/p06': () => const _FormTwo(id: 'p06', focus: _Focus.input),
  '/p07': () => const _FormTwo(id: 'p07', focus: _Focus.semanticsEvent),
  '/p08': () => const _FormHint(),
  '/p09': () => const _CupertinoRow(),
  '/p10': () => const _WidgetsLayer(),
  '/p11': () => const _FormSummary(),
  '/p12': () => const _AutovalidateWhileTyping(),
  '/p13': () => const _CupertinoHandBuilt(),
  // Second round, after the first showed what MergeSemantics does to a
  // field's error: the identifier straight on the field, not merged.
  '/q01': () => const _Unmerged(id: 'q01', focus: false, filled: false),
  '/q02': () => const _Unmerged(id: 'q02', focus: true, filled: false),
  '/q03': () => const _Unmerged(id: 'q03', focus: true, filled: true),
  '/q04': () => const _CupertinoCandidate(),
  '/q05': () => const _DetachedError(),
  '/q06': () => const _ColorOnly(),
};

/// q01 to q03: Form with Name and Email, identifiers by Semantics(identifier:)
/// only. q03 starts with Name "Ana" and Email "abc".
class _Unmerged extends StatefulWidget {
  const _Unmerged({required this.id, required this.focus, required this.filled});

  final String id;
  final bool focus;
  final bool filled;

  @override
  State<_Unmerged> createState() => _UnmergedState();
}

class _UnmergedState extends State<_Unmerged> {
  final _form = GlobalKey<FormState>();
  final _nameKey = GlobalKey<FormFieldState<String>>();
  final _nameFocus = FocusNode();
  final _emailFocus = FocusNode();
  late final _name = TextEditingController(text: widget.filled ? 'Ana' : '');
  late final _mail = TextEditingController(text: widget.filled ? 'abc' : '');

  @override
  void dispose() {
    _nameFocus.dispose();
    _emailFocus.dispose();
    _name.dispose();
    _mail.dispose();
    super.dispose();
  }

  void _submit() {
    if (_form.currentState!.validate() || !widget.focus) return;
    (_nameKey.currentState!.hasError ? _nameFocus : _emailFocus).requestFocus();
  }

  @override
  Widget build(BuildContext context) {
    final p = widget.id;
    return Form(
      key: _form,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          Semantics(
            identifier: '$p-name',
            child: TextFormField(
              key: _nameKey,
              controller: _name,
              focusNode: _nameFocus,
              decoration: InputDecoration(
                labelText: '$p Name',
                border: const OutlineInputBorder(),
              ),
              validator: (v) => _required(v, _nameError),
            ),
          ),
          _gap(),
          Semantics(
            identifier: '$p-email',
            child: TextFormField(
              controller: _mail,
              focusNode: _emailFocus,
              decoration: InputDecoration(
                labelText: '$p Email',
                border: const OutlineInputBorder(),
              ),
              validator: _email,
            ),
          ),
          _gap(),
          TestId(
            '$p-submit',
            child: FilledButton(onPressed: _submit, child: Text('$p Submit')),
          ),
        ],
      ),
    );
  }
}

/// q04: a Cupertino field built the way InputDecorator does it: the error
/// is the field's hint and its validation result, and the visible error is a
/// live region only where announcements are not supported. Focus moves to
/// the field.
class _CupertinoCandidate extends StatefulWidget {
  const _CupertinoCandidate();

  @override
  State<_CupertinoCandidate> createState() => _CupertinoCandidateState();
}

class _CupertinoCandidateState extends State<_CupertinoCandidate> {
  final _form = GlobalKey<FormState>();
  final _focus = FocusNode();

  @override
  void dispose() {
    _focus.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Form(
      key: _form,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          FormField<String>(
            validator: _email,
            builder: (field) => CupertinoFormSection.insetGrouped(
              children: [
                CupertinoFormRow(
                  prefix: const ExcludeSemantics(child: Text('q04 Email')),
                  error: field.errorText == null
                      ? null
                      : Semantics(
                          container: true,
                          liveRegion: !MediaQuery.supportsAnnounceOf(context),
                          child: Text(field.errorText!),
                        ),
                  child: Semantics(
                    identifier: 'q04-email',
                    label: 'q04 Email',
                    hint: field.errorText,
                    validationResult: field.hasError
                        ? SemanticsValidationResult.invalid
                        : SemanticsValidationResult.valid,
                    child: CupertinoTextField.borderless(
                      focusNode: _focus,
                      placeholder: 'name@example.com',
                      onChanged: field.didChange,
                    ),
                  ),
                ),
              ],
            ),
          ),
          _gap(),
          TestId(
            'q04-submit',
            child: CupertinoButton.filled(
              onPressed: () {
                if (!_form.currentState!.validate()) _focus.requestFocus();
              },
              child: const Text('q04 Submit'),
            ),
          ),
        ],
      ),
    );
  }
}

enum _Focus { none, input, semanticsEvent }

/// p05 to p07: Form with two required fields; Submit validates, then moves
/// nothing (p05), input focus (p06), or sends a FocusSemanticEvent (p07)
/// to the first field in error.
class _FormTwo extends StatefulWidget {
  const _FormTwo({required this.id, required this.focus});

  final String id;
  final _Focus focus;

  @override
  State<_FormTwo> createState() => _FormTwoState();
}

class _FormTwoState extends State<_FormTwo> {
  final _form = GlobalKey<FormState>();
  final _nameKey = GlobalKey<FormFieldState<String>>();
  final _emailKey = GlobalKey<FormFieldState<String>>();
  final _nameFocus = FocusNode();
  final _emailFocus = FocusNode();

  @override
  void dispose() {
    _nameFocus.dispose();
    _emailFocus.dispose();
    super.dispose();
  }

  void _submit() {
    final ok = _form.currentState!.validate();
    if (ok) return;
    final first = _nameKey.currentState!.hasError ? _nameKey : _emailKey;
    switch (widget.focus) {
      case _Focus.none:
        break;
      case _Focus.input:
        (first == _nameKey ? _nameFocus : _emailFocus).requestFocus();
      case _Focus.semanticsEvent:
        WidgetsBinding.instance.addPostFrameCallback((_) {
          first.currentContext?.findRenderObject()?.sendSemanticsEvent(
            const FocusSemanticEvent(),
          );
        });
    }
  }

  @override
  Widget build(BuildContext context) {
    final p = widget.id;
    return Form(
      key: _form,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          TestId(
            '$p-name',
            child: TextFormField(
              key: _nameKey,
              focusNode: _nameFocus,
              decoration: InputDecoration(
                labelText: '$p Name',
                border: const OutlineInputBorder(),
              ),
              validator: (v) => _required(v, _nameError),
            ),
          ),
          _gap(),
          TestId(
            '$p-email',
            child: TextFormField(
              key: _emailKey,
              focusNode: _emailFocus,
              decoration: InputDecoration(
                labelText: '$p Email',
                border: const OutlineInputBorder(),
              ),
              validator: _email,
            ),
          ),
          _gap(),
          TestId(
            '$p-submit',
            child: FilledButton(
              onPressed: _submit,
              child: Text('$p Submit'),
            ),
          ),
        ],
      ),
    );
  }
}

/// p08: the error is also the field's hint (the iOS reference's
/// accessibilityHint). Phone holds "abc".
class _FormHint extends StatefulWidget {
  const _FormHint();

  @override
  State<_FormHint> createState() => _FormHintState();
}

class _FormHintState extends State<_FormHint> {
  final _form = GlobalKey<FormState>();
  final _phone = TextEditingController(text: 'abc');
  String? _emailErr;
  String? _phoneErr;

  @override
  void dispose() {
    _phone.dispose();
    super.dispose();
  }

  void _submit() {
    setState(() {
      _emailErr = _email('');
      _phoneErr = RegExp(r'^\d+$').hasMatch(_phone.text) ? null : _phoneError;
    });
    _form.currentState!.validate();
  }

  @override
  Widget build(BuildContext context) {
    return Form(
      key: _form,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          TestId(
            'p08-email',
            child: Semantics(
              hint: _emailErr,
              child: TextFormField(
                decoration: const InputDecoration(
                  labelText: 'p08 Email',
                  border: OutlineInputBorder(),
                ),
                validator: _email,
              ),
            ),
          ),
          _gap(),
          TestId(
            'p08-phone',
            child: Semantics(
              hint: _phoneErr,
              child: TextFormField(
                controller: _phone,
                decoration: const InputDecoration(
                  labelText: 'p08 Phone',
                  border: OutlineInputBorder(),
                ),
                validator: (v) =>
                    RegExp(r'^\d+$').hasMatch(v ?? '') ? null : _phoneError,
              ),
            ),
          ),
          _gap(),
          TestId(
            'p08-submit',
            child: FilledButton(
              onPressed: _submit,
              child: const Text('p08 Submit'),

          ),
        ],
      ),
    );
  }
