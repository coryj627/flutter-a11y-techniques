"""F06 probe runner for Android (throwaway; kept as
docs/evidence/probes/f06/probe_android.py.txt). Run under the device lock:

  python3 tool/with_devices.py --label "probe F06 android" -- python3 <this> <repo> [tree|talkback|both]

Runs ProbeF06Test against lib/probe.dart. "tree" runs probe() without TalkBack
and saves the F06PROBE log lines; "talkback" turns TalkBack on with the
project's own helpers (tool/talkback.py: run_with_talkback, which restores
the device), runs talkBackWalk(), and saves the speech per marker as parsed
by tool/talkback.py's parse_speech, with the F06PROBE lines.
"""
import os
import subprocess
import sys
import time

ROOT = sys.argv[1]
MODE = sys.argv[2] if len(sys.argv) > 2 else "both"
# Round "2" runs probe2 and talkBackWalk2 and writes files ending in 2.
ROUND = sys.argv[3] if len(sys.argv) > 3 else ""
OUT = os.path.join(ROOT, "docs", "evidence", "probes", "f06")
sys.path.insert(0, os.path.join(ROOT, "tool"))
import talkback  # noqa: E402
from evidence_common import GRADLE_UNSHARED, stop_android_tests  # noqa: E402

PKG = "dev.fluttera11y.flutter_a11y_techniques"


def gradle(method, log_path):
    talkback.adb("logcat", "-c")
    with open(log_path, "w") as log:
        reader = subprocess.Popen([talkback.ADB, "logcat"], stdout=log,
                                  stderr=subprocess.DEVNULL)
        try:
            run = subprocess.run(
                ["./gradlew", *GRADLE_UNSHARED, "app:connectedDebugAndroidTest",
                 "-Ptarget=lib/probe.dart",
                 f"-Pandroid.testInstrumentationRunnerArguments.class={PKG}.ProbeF06Test#{method}"],
                cwd=os.path.join(ROOT, "android"), capture_output=True, text=True, timeout=40 * 60)
        finally:
            time.sleep(2)
            reader.terminate()
            reader.wait()
    print(run.stdout[-2500:])
    print(run.stderr[-2500:])
    with open(log_path) as f:
        return run, f.read()


def keep(log, path):
    lines = [l for l in log.splitlines() if "F06PROBE" in l or "F06TB" in l]
    with open(path, "w") as f:
        f.write("\n".join(lines) + "\n")


def main():
    os.makedirs(OUT, exist_ok=True)
    scratch = os.path.dirname(os.path.abspath(__file__))
    if not stop_android_tests():
        sys.exit("emulator not clean")
    if MODE in ("tree", "both"):
        run, log = gradle("probe" + ROUND, os.path.join(scratch, f"android-tree{ROUND}-logcat.txt"))
        keep(log, os.path.join(OUT, f"android-tree{ROUND}.txt"))
        stop_android_tests()
    if MODE in ("talkback", "both"):
        talkback.exit_on_termination_signals()

        def action():
            return gradle("talkBackWalk" + ROUND, os.path.join(scratch, f"android-talkback{ROUND}-logcat.txt"))

        run, log = talkback.run_with_talkback(action)
        stop_android_tests()
        spoken = talkback.parse_speech(log)
        out = [f"| Marker | TalkBack spoke |", "| --- | --- |"]
        for ident, fragments in spoken.items():
            out.append(f"| `{ident}` | {' / '.join(fragments) or '(nothing)'} |")
        with open(os.path.join(OUT, f"android-talkback{ROUND}.md"), "w") as f:
            f.write("\n".join(out) + "\n")
        keep(log, os.path.join(OUT, f"android-talkback-tree{ROUND}.txt"))
        print("\n".join(out))


if __name__ == "__main__":
    main()
