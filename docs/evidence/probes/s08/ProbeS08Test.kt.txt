// S08 probe (throwaway, not committed; kept as
// docs/evidence/probes/s08/ProbeS08Test.kt.txt). Runs against lib/probe.dart
// (Gradle -Ptarget=lib/probe.dart). probe() dumps every node of each
// fixture with its collection information; for the lazy fixtures (p08,
// p10) it scrolls the inner list forward with ACTION_SCROLL_FORWARD, one
// step at a time, and dumps the item nodes after each step. nativeTree()
// dumps the native comparison (ProbeNativeActivity). talkBackWalk() moves
// TalkBack focus to "Before", to each node of the fixture with a name, and
// to "end"; talkBackKeys() puts TalkBack focus on "Before" and presses
// TalkBack's keyboard shortcut for the next item (Alt+Right) up to twelve
// times, with a marker before each press.
package dev.fluttera11y.flutter_a11y_techniques

import android.content.Intent
import android.graphics.Rect
import android.util.Log
import android.view.accessibility.AccessibilityNodeInfo
import android.view.accessibility.AccessibilityWindowInfo
import androidx.test.ext.junit.runners.AndroidJUnit4
import org.junit.Test
import org.junit.runner.RunWith

@RunWith(AndroidJUnit4::class)
class ProbeS08Test : TechniqueTest("S08") {
    private val tag = "S08PROBE"
    private val ids = (1..29).map { "p%02d".format(it) }

    private fun launch(fixture: String) {
        val intent = Intent(inst.targetContext, MainActivity::class.java)
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK)
            .putExtra("route", "/$fixture")
        inst.targetContext.startActivity(intent)
        val end = System.currentTimeMillis() + 20_000
        while (!anyWindowHas { label(it) == "Before $fixture" }) {
            if (System.currentTimeMillis() > end) {
                Log.i(tag, "$fixture did not open (no \"Before $fixture\" node)")
                dump("$fixture not open")
                return
            }
            Thread.sleep(250)
        }
        device.waitForIdle()
        Thread.sleep(1_000)
    }

    private fun openNative() {
        val intent = Intent(inst.targetContext, ProbeNativeActivity::class.java)
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK)
        inst.targetContext.startActivity(intent)
        val end = System.currentTimeMillis() + 20_000
        while (!anyWindowHas { it.text?.toString() == "Before n01" }) {
            if (System.currentTimeMillis() > end) throw AssertionError("native screen did not open")
            Thread.sleep(250)
        }
        device.waitForIdle()
        Thread.sleep(1_000)
    }

    private fun describe(n: AccessibilityNodeInfo): String {
        val r = Rect().also { n.getBoundsInScreen(it) }
        val actions = n.actionList.joinToString(",") { a ->
            when (a.id) {
                AccessibilityNodeInfo.ACTION_CLICK -> "click"
                AccessibilityNodeInfo.ACTION_FOCUS -> "focus"
                AccessibilityNodeInfo.ACTION_ACCESSIBILITY_FOCUS -> "a11yfocus"
                AccessibilityNodeInfo.ACTION_CLEAR_ACCESSIBILITY_FOCUS -> "clra11y"
                AccessibilityNodeInfo.ACTION_SCROLL_FORWARD -> "scrollfwd"
                AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD -> "scrollback"
                else -> a.id.toString()
            }
        }
        val ci = n.collectionInfo?.let { " collection=${it.rowCount}x${it.columnCount} hierarchical=${it.isHierarchical}" } ?: ""
        val cii = n.collectionItemInfo?.let { " item=row${it.rowIndex},col${it.columnIndex} itemHeading=${it.isHeading}" } ?: ""
        return "${n.viewIdResourceName ?: "-"} | ${n.type()} | desc=\"${n.name().replace("\n", "\\n")}\" " +
            "text=\"${(n.text ?: "").toString().replace("\n", "\\n")}\" heading=${n.isHeading} " +
            "clickable=${n.isClickable} focusable=${n.isFocusable} scrollable=${n.isScrollable} " +
            "visible=${n.isVisibleToUser} $r [$actions] children=${n.childCount}$ci$cii"
    }

    private fun collect(root: AccessibilityNodeInfo?, depth: Int, out: MutableList<Pair<Int, AccessibilityNodeInfo>>) {
        if (root == null) return
        out.add(depth to root)
        for (i in 0 until root.childCount) collect(root.getChild(i), depth + 1, out)
    }

    private fun appNodes(): List<Pair<Int, AccessibilityNodeInfo>> {
        val out = mutableListOf<Pair<Int, AccessibilityNodeInfo>>()
        for (w in automation.windows) if (w.type == AccessibilityWindowInfo.TYPE_APPLICATION) collect(w.root, 0, out)
        return out
    }

    private fun dump(label: String) {
        Log.i(tag, "==== $label")
        for ((depth, n) in appNodes()) Log.i(tag, "${"  ".repeat(depth)}${describe(n)}")
    }

    private fun label(n: AccessibilityNodeInfo) = n.name().ifEmpty { n.text?.toString().orEmpty() }

    /** "Before", each node of the fixture with a name, "end". */
    private fun stops(id: String) = appNodes().map { it.second }.filter {
        val l = label(it)
        l == "Before $id" || l.startsWith("$id ")
    }

    @Test
    fun probe() {
        for (id in ids) {
            launch(id)
            dump("$id start")
            if (id == "p08" || id == "p10") {
                for (step in 1..6) {
                    val list = appNodes().map { it.second }.lastOrNull { it.isScrollable && it.childCount > 0 &&
                        (0 until it.childCount).any { i -> it.getChild(i)?.let { c -> label(c).startsWith("$id Item") } == true } }
                    if (list == null) {
                        Log.i(tag, "$id: no scrollable list node found")
                        break
                    }
                    val ok = list.performAction(AccessibilityNodeInfo.ACTION_SCROLL_FORWARD)
                    Thread.sleep(1_200)
                    Log.i(tag, "$id ACTION_SCROLL_FORWARD step $step returned $ok")
                    for ((_, n) in appNodes()) {
                        if (n.isScrollable || label(n).startsWith("$id Item")) Log.i(tag, "$id step $step ${describe(n)}")
                    }
                }
            }
        }
    }

    @Test
    fun nativeTree() {
        openNative()
        dump("native")
    }

    private fun a11yFocused(): String {
        val f = automation.windows.firstNotNullOfOrNull { it.root?.findFocus(AccessibilityNodeInfo.FOCUS_ACCESSIBILITY) }
        return f?.let { label(it).replace("\n", "\\n").ifEmpty { "(${it.type()} no name)" } } ?: "(none)"
    }

    private fun walkStops(prefix: String, nodes: () -> List<AccessibilityNodeInfo>) {
        val parts = nodes()
        parts.forEachIndexed { i, n ->
            val marker = "s08-$prefix-$i"
            Log.i(tag, "$marker is \"${label(n).replace("\n", "\\n")}\"")
            Log.i(talkBackTag, "FOCUS $marker")
            try {
                talkBackFocus(marker) { n }
            } catch (e: AssertionError) {
                Log.i(tag, "$marker: TalkBack focus refused")
            }
            Thread.sleep(5_000)
        }
        Log.i(talkBackTag, "END $prefix")
    }

    @Test
    fun talkBackWalk() {
        assumeTalkBack()
        openNative()
        Thread.sleep(6_000)
        walkStops("native") {
            appNodes().map { it.second }.filter { val l = label(it); l.startsWith("n0") || l.startsWith("Before n0") }
        }
        for (id in ids) {
            launch(id)
            Thread.sleep(6_000)
            walkStops(id) { stops(id) }
        }
    }

    private fun keyWalk(prefix: String, first: () -> AccessibilityNodeInfo?) {
        val start = first() ?: run { Log.i(tag, "$prefix: no first node"); return }
        talkBackFocus("$prefix start") { start }
        Thread.sleep(3_000)
        Log.i(tag, "$prefix k-start on ${a11yFocused()}")
        for (i in 1..12) {
            val marker = "s08-k$prefix-$i"
            Log.i(talkBackTag, "FOCUS $marker")
            device.executeShellCommand("input keycombination KEYCODE_ALT_LEFT KEYCODE_DPAD_RIGHT")
            Thread.sleep(3_000)
            Log.i(tag, "$marker after Alt+Right: TalkBack focus on ${a11yFocused()}")
        }
        Log.i(talkBackTag, "END k$prefix")
    }

    @Test
    fun talkBackKeys() {
        assumeTalkBack()
        openNative()
        Thread.sleep(6_000)
        keyWalk("native") { appNodes().map { it.second }.firstOrNull { it.text?.toString() == "Before n01" } }
        for (id in listOf("p01", "p03", "p06", "p08", "p13", "p24", "p28")) {
            launch(id)
            Thread.sleep(6_000)
            keyWalk(id) { appNodes().map { it.second }.firstOrNull { label(it) == "Before $id" } }
        }
    }

    // Round 2: the items the first walk missed (their text starts with a
    // bullet or a number: the native n02 and n03, and p01 and p15), and the
    // lazy lists (p08, and the native n04) walked again after their list is
    // scrolled forward three steps with ACTION_SCROLL_FORWARD.

    /** The scrollable node that holds an item whose label starts with [prefix]. */
    private fun scrollerOf(prefix: String): AccessibilityNodeInfo? = appNodes().map { it.second }.lastOrNull { n ->
        n.isScrollable && (0 until n.childCount).any { i -> n.getChild(i)?.let { c -> label(c).startsWith(prefix) } == true }
    }

    private fun scrolledWalk(prefix: String, item: String) {
        for (step in 1..3) {
            val ok = scrollerOf(item)?.performAction(AccessibilityNodeInfo.ACTION_SCROLL_FORWARD)
            Thread.sleep(1_500)
            Log.i(tag, "$prefix ACTION_SCROLL_FORWARD step $step returned $ok")
        }
        val visible = appNodes().map { it.second }.filter { label(it).startsWith(item) && it.isVisibleToUser }
        Log.i(tag, "$prefix visible after scrolling: ${visible.joinToString(",") { label(it) }}")
        walkStops("$prefix-scrolled") { visible }
    }

    @Test
    fun talkBackRound2() {
        assumeTalkBack()
        openNative()
        Thread.sleep(6_000)
        walkStops("native2") {
            appNodes().map { it.second }.filter { val l = label(it); l.contains("n02") || l.contains("n03") }
        }
        walkStops("native4") {
            appNodes().map { it.second }.filter { label(it).contains("n04") && it.isVisibleToUser }
        }
        scrolledWalk("native4", "n04 Item")
        for (id in listOf("p01", "p15", "p08")) {
            launch(id)
            Thread.sleep(6_000)
            walkStops("r2-$id") {
                appNodes().map { it.second }.filter { val l = label(it); l == "Before $id" || l.contains("$id ") }
            }
            if (id == "p08") scrolledWalk("r2-p08", "p08 Item")
        }
    }
}
