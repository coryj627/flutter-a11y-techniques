// S08 probe (throwaway, not committed; kept as
// docs/evidence/probes/s08/probe.dart.txt): what lists built from the SDK
// expose. One fixture per route, `/p01` to `/p22`; `/` lists nothing.
//
// Each fixture sits on a screen laid out as the app's own technique screens
// are (a `Scaffold` whose body is a `SingleChildScrollView` round a
// `Column`), between a "Before pNN" text and a "pNN end" text, which has the
// test identifier `pNN-end`. Every item's text starts with the fixture's id.
import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';

void main() {
  runApp(const ProbeApp());
  SemanticsBinding.instance.ensureSemantics();
}

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      onGenerateRoute: (settings) {
        final name = (settings.name ?? '/').replaceFirst('/', '');
        final page = fixtures.containsKey(name)
            ? FixtureScreen(id: name)
            : const Scaffold(body: Text('S08 probe'));
        return MaterialPageRoute<void>(
          settings: settings,
          builder: (_) => page,
        );
      },
    );
  }
}

typedef Build = Widget Function(String id, void Function() tap);

const groceries = ['Milk', 'Eggs', 'Bread'];
const steps = ['Fork the repository', 'Make the change', 'Open a pull request'];

List<Widget> texts(String id, {String bullet = ''}) => [
  for (final g in groceries) Text('$bullet$id $g'),
];

Widget item(Widget child) =>
    Semantics(role: SemanticsRole.listItem, child: child);

Widget list(Widget child) => Semantics(role: SemanticsRole.list, child: child);

Widget column(List<Widget> children) => Column(
  crossAxisAlignment: CrossAxisAlignment.start,
  mainAxisSize: MainAxisSize.min,
  children: children,
);

ListView shrunk(List<Widget> children) => ListView(
  shrinkWrap: true,
  physics: const NeverScrollableScrollPhysics(),
  children: children,
);

Widget lazy(String id, {bool roles = false}) => SizedBox(
  height: 200,
  child: ListView.builder(
    itemCount: 30,
    itemBuilder: (context, i) {
      final t = Padding(
        padding: const EdgeInsets.symmetric(vertical: 8),
        child: Text('$id Item ${i + 1}'),
      );
      return roles ? item(t) : t;
    },
  ),
);

class FixtureScreen extends StatefulWidget {
  const FixtureScreen({super.key, required this.id});

  final String id;

  @override
  State<FixtureScreen> createState() => _FixtureScreenState();
}

class _FixtureScreenState extends State<FixtureScreen> {
  var taps = 0;

  @override
  Widget build(BuildContext context) {
    final id = widget.id;
    return Scaffold(
      appBar: AppBar(title: Text('Probe $id')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          spacing: 12,
          children: [
            Text('Before $id'),
            fixtures[id]!.$2(id, () => setState(() => taps++)),
            Semantics(identifier: '$id-end', child: Text('$id end taps=$taps')),
          ],
        ),
      ),
    );
  }
}
