// S08 probe (throwaway, not committed; kept as
// docs/evidence/probes/s08/ProbeS08UITests.swift.txt): for each fixture of
// lib/probe.dart, opened by URL: every element of one snapshot whose label
// or identifier has the fixture's id; for the lazy fixtures (p08, p10), the
// list swiped up a step at a time and the item elements after each step.
// With VoiceOver on: a walk forward from the top of each fixture to its end
// line (testVoiceOver). testSafari: plain HTML lists in Safari, served from
// the Mac, walked forward with VoiceOver, as a comparison.
import XCTest

final class ProbeS08UITests: XCTestCase {
  let app = XCUIApplication()
  let ids = (1...29).map { String(format: "p%02d", $0) }

  func log(_ s: String) { print("S08PROBE \(s)") }

  func el(_ id: String) -> XCUIElement { app.descendants(matching: .any)[id] }

  @discardableResult
  func open(_ id: String) -> Bool {
    for _ in 1...3 {
      app.terminate()
      app.open(URL(string: "fluttera11y:///\(id)")!)
      if el("\(id)-end").waitForExistence(timeout: 30) { sleep(1); return true }
    }
    log("\(id) did not open")
    return false
  }

  func dump(_ id: String, _ label: String, itemsOnly: Bool = false) {
    guard let root = try? app.snapshot() else { log("\(id) \(label): no snapshot"); return }
    var out: [String] = []
    func walk(_ s: XCUIElementSnapshot) {
      let l = s.label
      let mine = l.contains(id) || s.identifier.hasPrefix(id)
      if mine && s.elementType != .window && s.elementType != .application
        && (!itemsOnly || l.contains("Item")) {
        out.append("type=\(s.elementType.rawValue) id=\(s.identifier) label=\(l.debugDescription) value=\(String(describing: s.value)) frame=\(s.frame)")
      }
      for c in s.children { walk(c) }
    }
    walk(root)
    log("\(id) \(label): \(out.count) element(s): \(out.joined(separator: " | "))")
  }

  /// Round 2: the fixtures testTree did not reach in round 1, after its
  /// drag on p08 failed (the element dragged from had scrolled away).
  func testTree2() {
    tree(["p08"] + (11...29).map { String(format: "p%02d", $0) })
  }

  func testTree() { tree(ids) }

  func tree(_ ids: [String]) {
    for id in ids {
      guard open(id) else { continue }
      dump(id, "start")
      if id == "p08" || id == "p10" {
        let first = app.staticTexts["\(id) Item 2"]
        guard first.exists else { log("\(id): no Item 2 element"); continue }
        // A point on the screen, not on the element, which scrolls away.
        let frame = first.frame
        let start = app.coordinate(withNormalizedOffset: .zero)
          .withOffset(CGVector(dx: frame.midX, dy: frame.midY))
        for step in 1...4 {
          start.press(forDuration: 0.05, thenDragTo: start.withOffset(CGVector(dx: 0, dy: -150)))
          sleep(1)
          dump(id, "after drag \(step)", itemsOnly: true)
        }
      }
    }
  }

  func testVoiceOver() throws {
    #if compiler(>=6.4)
    guard #available(iOS 27.0, *) else { throw XCTSkip("needs iOS 27") }
    let vo = XCUIDevice.shared.voiceOverService
    func norm(_ s: String) -> String { s.split(whereSeparator: { $0.isWhitespace }).joined(separator: " ") }
    func say() -> String {
      do { return norm(try vo.moveForward().utterance) } catch { return "(error \((error as NSError).code))" }
    }
    for id in ids {
      guard open(id) else { continue }
      try vo.enable()
      var walk: [String] = []
      let limit = (id == "p08" || id == "p10") ? 45 : 16
      for _ in 0..<limit {
        let s = say()
        walk.append(s)
        if s.hasPrefix("\(id) end") { break }
      }
      log("VO \(id) walk: \(walk.joined(separator: " / "))")
      try vo.disable()
    }
    #else
    throw XCTSkip("needs Xcode 27")
    #endif
  }

  func testSafari() throws {
    #if compiler(>=6.4)
    guard #available(iOS 27.0, *) else { throw XCTSkip("needs iOS 27") }
    let port = ProcessInfo.processInfo.environment["S08_HTML_PORT"] ?? "29879"
    let safari = XCUIApplication(bundleIdentifier: "com.apple.mobilesafari")
    let vo = XCUIDevice.shared.voiceOverService
    func norm(_ s: String) -> String { s.split(whereSeparator: { $0.isWhitespace }).joined(separator: " ") }
    func say() -> String {
      do { return norm(try vo.moveForward().utterance) } catch { return "(error \((error as NSError).code))" }
    }
    for page in ["ul", "ol", "dl", "nested"] {
      safari.terminate()
      safari.open(URL(string: "http://127.0.0.1:\(port)/\(page).html")!)
      let start = safari.webViews.staticTexts["Before \(page)"]
      if !start.waitForExistence(timeout: 30) {
        log("safari \(page): page did not load; texts: \(safari.staticTexts.allElementsBoundByIndex.prefix(10).map { $0.label })")
        continue
      }
      sleep(1)
      // Round 1 read every element of the page here and failed when the
      // page changed under the read; round 2 reads one snapshot.
      if let snap = try? safari.webViews.firstMatch.snapshot() {
        var out: [String] = []
        func walk(_ s: XCUIElementSnapshot) {
          if !s.label.isEmpty { out.append("type=\(s.elementType.rawValue) \(s.label.debugDescription)") }
          for c in s.children { walk(c) }
        }
        walk(snap)
        log("safari \(page) elements: \(out.joined(separator: " | "))")
      }
      try vo.enable()
      start.tap()
      sleep(1)
      var walk: [String] = [norm((try? vo.currentSpeech().utterance) ?? "(none)")]
      for _ in 0..<14 {
        let s = say()
        walk.append(s)
        if s.hasPrefix("\(page) end") { break }
      }
      log("safari VO \(page) walk from a tap on Before: \(walk.joined(separator: " / "))")
      try vo.disable()
    }
    safari.terminate()
    #else
    throw XCTSkip("needs Xcode 27")
    #endif
  }
}
