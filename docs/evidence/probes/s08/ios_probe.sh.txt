#!/bin/sh
# S08 probe on iOS, under tool/with_devices.py (throwaway; kept as
# docs/evidence/probes/s08/ios_probe.sh.txt). Run from the worktree root,
# after `fvm flutter build ios --simulator --debug -t lib/probe.dart` and
# `xcodebuild build-for-testing` into build/ios_uitest. Serves the plain
# HTML pages for testSafari from $HTML on 127.0.0.1:29879 for the run.
# Changes no simulator setting; testVoiceOver and testSafari turn
# VoiceOver on and off through XCUIVoiceOverService.
xcrun simctl list devices booted | grep -i "iphone"
python3 -m http.server 29879 --bind 127.0.0.1 --directory "$HTML" > /dev/null 2>&1 &
SERVER=$!
trap 'kill $SERVER 2>/dev/null' EXIT
cd ios || exit 1
run() {
  echo "== $1"
  xcodebuild test-without-building -workspace Runner.xcworkspace -scheme Runner \
    -destination "platform=iOS Simulator,name=iPhone 17" \
    -only-testing:"RunnerUITests/ProbeS08UITests/$1" \
    -derivedDataPath ../build/ios_uitest 2>&1 | grep --line-buffered -E "S08PROBE|error:|passed|failed|skipped" | grep --line-buffered -v "Test Suite"
}
for t in ${TESTS:-testTree testVoiceOver testSafari}; do run $t; done
xcrun simctl terminate booted dev.fluttera11y.flutterA11yTechniques 2>/dev/null
xcrun simctl terminate booted com.apple.mobilesafari 2>/dev/null
exit 0
