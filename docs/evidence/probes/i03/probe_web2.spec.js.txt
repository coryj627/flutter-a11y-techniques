// Throwaway: I03 probe round 2 on the web (Chromium). Not committed; kept
// as docs/evidence/probes/i03/probe_web2.spec.js.txt.
import { test } from '@playwright/test';

const id = (page, x) => page.locator(`[flt-semantics-identifier="${x}"]`);
const btn = (page, x) => page.getByRole('button', { name: x, exact: true });
const status = async (page, x) => (await id(page, x).getAttribute('aria-label')) ?? (await id(page, x).textContent()) ?? '';
const active = (page) => page.evaluate(() => {
  const e = document.activeElement;
  if (!e || e === document.body) return 'body';
  return (e.getAttribute('aria-label') || e.textContent || e.tagName).trim().slice(0, 30);
});
const kf = async (page) => (await status(page, 'kf')).replace('KF: ', '');
const where = async (page) => `KF ${await kf(page)} / DOM ${await active(page)}`;

async function press(page, key) {
  await page.keyboard.press(key);
  await page.waitForTimeout(250);
}

async function tabTo(page, text, max = 40) {
  for (let i = 0; i < max; i++) {
    if ((await kf(page)) === `node:${text}`) return true;
    await press(page, 'Tab');
  }
  return (await kf(page)) === `node:${text}`;
}

async function round2(page) {
  await page.goto('about:blank');
  await page.goto('/#/round2');
  await id(page, 'r9 Last').waitFor();
  await page.waitForTimeout(800);
}

for (const list of ['r1', 'r2', 'r3', 'r5']) {
  test(`keyboard ${list}`, async ({ page }) => {
    await round2(page);
    await tabTo(page, `${list} Remove A`);
    await press(page, 'Enter');
    await page.waitForTimeout(500);
    const open = await where(page);
    await tabTo(page, `${list} Cancel`, 6);
    await press(page, 'Enter');
    await page.waitForTimeout(1500);
    const afterCancel = await where(page);
    await tabTo(page, `${list} Remove B`);
    await press(page, 'Enter');
    await page.waitForTimeout(500);
    await tabTo(page, `${list} Confirm`, 6);
    await press(page, 'Enter');
    await page.waitForTimeout(2500);
    const afterConfirm = await where(page);
    await press(page, 'Tab');
    console.log(`I03PROBE web round2 keyboard ${list} | after Enter: ${open} | after Cancel on A: ${afterCancel} | after Confirm on B: ${afterConfirm} | Tab: ${await where(page)}`);
  });

  // As VoiceOver does: DOM focus on the element, then a click.
  test(`focus and click ${list}`, async ({ page }) => {
    await round2(page);
    const act = async (name) => {
      await btn(page, name).focus();
      await page.waitForTimeout(300);
      await btn(page, name).click();
      await page.waitForTimeout(1500);
    };
    await act(`${list} Remove A`);
    const open = await where(page);
    await act(`${list} Cancel`);
    const afterCancel = await where(page);
    await act(`${list} Remove B`);
    await act(`${list} Confirm`);
    await page.waitForTimeout(1500);
    const afterConfirm = await where(page);
    await press(page, 'Tab');
    console.log(`I03PROBE web round2 click ${list} | after opening A: ${open} | after Cancel: ${afterCancel} | after Confirm on B: ${afterConfirm} | Tab: ${await where(page)}`);
  });
}

test('r4 requestFocus false', async ({ page }) => {
  await round2(page);
  await tabTo(page, 'r4 Dialog without focus');
  await press(page, 'Enter');
  await page.waitForTimeout(800);
  const open = await where(page);
  const path = [];
  for (let i = 0; i < 4; i++) {
    await press(page, 'Tab');
    path.push(await where(page));
  }
  const behind = await btn(page, 'r9 Last').count();
  console.log(`I03PROBE web round2 r4 | after Enter: ${open} | Tab x4: ${path.join(' > ')} | r9 Last in the DOM while open: ${behind}`);
});

async function probe(page) {
  await page.goto('about:blank');
  await page.goto('/#/probe');
  await id(page, 'p16 Last').waitFor();
  await page.waitForTimeout(800);
}

test('Enter on a CupertinoActionSheetAction', async ({ page }) => {
  await probe(page);
  await tabTo(page, 'p6 Action sheet');
  await press(page, 'Enter');
  await page.waitForTimeout(800);
  await tabTo(page, 'p6 Close', 6);
  await press(page, 'Enter');
  await page.waitForTimeout(1000);
  const afterEnter = `${await where(page)} | sheet open: ${await btn(page, 'p6 Close').count()}`;
  await press(page, 'Space');
  await page.waitForTimeout(1000);
  console.log(`I03PROBE web p6 Enter on p6 Close: ${afterEnter} | then Space: ${await where(page)} | sheet open: ${await btn(page, 'p6 Close').count()}`);
});

for (const [trigger, last] of [['p3 Sheet', 'p3 Close'], ['p6 Action sheet', 'p6 Close'], ['p8 General dialog', 'p8 Close'], ['p1 Dialog', 'p1 Close']]) {
  test(`Tab past the last control of ${trigger}`, async ({ page }) => {
    await probe(page);
    await tabTo(page, trigger);
    await press(page, 'Enter');
    await page.waitForTimeout(800);
    await tabTo(page, last, 6);
    const path = [];
    for (let i = 0; i < 4; i++) {
      await press(page, 'Tab');
      path.push(await where(page));
    }
    console.log(`I03PROBE web Tab past ${last}: ${path.join(' > ')}`);
  });
}

test('Escape in a CupertinoAlertDialog', async ({ page }) => {
  await probe(page);
  await tabTo(page, 'p5 Cupertino dialog');
  await press(page, 'Enter');
  await page.waitForTimeout(800);
  await press(page, 'Escape');
  await page.waitForTimeout(800);
  console.log(`I03PROBE web p5 Escape: ${await where(page)} | dialog open: ${await page.getByText('p5 Title').count()}`);
});
