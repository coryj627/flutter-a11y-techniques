#!/bin/sh
# I03 probe on Android, under tool/with_devices.py. Run from the worktree
# root. Runs the probe's instrumented tests without TalkBack, then turns
# TalkBack on (by the accessibility settings, as tool/talkback.py does) and
# runs the TalkBack ones, and puts both settings back as they were.
ADB=$HOME/Library/Android/sdk/platform-tools/adb
PKG=dev.fluttera11y.flutter_a11y_techniques
TB=com.google.android.marvin.talkback/com.google.android.marvin.talkback.TalkBackService
S0=$($ADB shell settings get secure enabled_accessibility_services | tr -d '\r')
E0=$($ADB shell settings get secure accessibility_enabled | tr -d '\r')
echo "== device: Android $($ADB shell getprop ro.build.version.release | tr -d '\r'), $($ADB shell wm size | tr -d '\r')"
echo "== settings before: enabled_accessibility_services=[$S0] accessibility_enabled=[$E0]"
put() { # name value; "null" deletes
  if [ "$2" = "null" ]; then $ADB shell settings delete secure "$1"; else $ADB shell settings put secure "$1" "${2:-\"\"}"; fi
}
restore() {
  put enabled_accessibility_services "$S0"
  put accessibility_enabled "$E0"
  echo "== settings restored: enabled_accessibility_services=[$($ADB shell settings get secure enabled_accessibility_services | tr -d '\r')] accessibility_enabled=[$($ADB shell settings get secure accessibility_enabled | tr -d '\r')]"
}
trap restore EXIT
run() {
  $ADB logcat -c
  (cd android && ./gradlew --no-daemon -Pkotlin.compiler.execution.strategy=in-process app:connectedDebugAndroidTest \
    -Ptarget=lib/probe.dart \
    -Pandroid.testInstrumentationRunnerArguments.class=$PKG.ProbeI03Test#$1 2>&1 | grep -E "FAILED|BUILD|tests completed|Exception|Assumption" )
  $ADB logcat -d -s I03PROBE:I | grep I03PROBE | sed 's/^.*I03PROBE: //'
}
echo "== without TalkBack"
run keyboard
run removalByKeyboard
run p15
echo "== with TalkBack"
put enabled_accessibility_services "$TB"
put accessibility_enabled 1
sleep 8
run talkBack
run talkBackRemoval
run talkBackWithKeyboard
run talkBackP15
exit 0
