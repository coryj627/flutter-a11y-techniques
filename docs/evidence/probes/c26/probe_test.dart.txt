// C26 probe widget test (throwaway; kept as
// docs/evidence/probes/c26/probe_test.dart.txt). Dumps every fixture's
// semantics as the framework builds it, as Android, iOS, and macOS (which
// a browser on a Mac reports), and every step of the changing fixtures.
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:flutter_a11y_techniques/probe.dart';

String describe(SemanticsNode n, int depth) {
  final d = n.getSemanticsData();
  final b = StringBuffer('${'  ' * depth}#${n.id}');
  if (d.identifier.isNotEmpty) b.write(' id=${d.identifier}');
  if (d.role != SemanticsRole.none) b.write(' role=${d.role.name}');
  if (d.label.isNotEmpty) b.write(' label="${d.label.replaceAll('\n', r'\n')}"');
  if (d.value.isNotEmpty) b.write(' value="${d.value}"');
  if (d.minValue != null) b.write(' min="${d.minValue}"');
  if (d.maxValue != null) b.write(' max="${d.maxValue}"');
  if (d.flagsCollection.isLiveRegion) b.write(' liveRegion');
  if (d.hasAction(SemanticsAction.tap)) b.write(' tap');
  b.writeln();
  n.visitChildren((c) {
    b.write(describe(c, depth + 1));
    return true;
  });
  return b.toString();
}

List<String> section(SemanticsNode root, String id) {
  final all = describe(root, 0).split('\n');
  final start = all.lastIndexWhere((l) => l.contains('"Before $id"'));
  final end = all.lastIndexWhere((l) => l.contains('$id end'));
  return all.sublist(start + 1, end);
}

void main() {
  for (final platform in [TargetPlatform.android, TargetPlatform.iOS, TargetPlatform.macOS]) {
    for (final id in fixtures.keys) {
      testWidgets('$platform $id', (tester) async {
        debugDefaultTargetPlatformOverride = platform;
        final handle = tester.ensureSemantics();
        await tester.pumpWidget(const ProbeApp());
        await tester.pump();
        tester.state<NavigatorState>(find.byType(Navigator)).pushNamed('/$id');
        await tester.pump();
        await tester.pump(const Duration(seconds: 1));
        final root = tester.binding.pipelineOwner.semanticsOwner!.rootSemanticsNode!;
        debugPrint('C26TEST ${platform.name} $id\n${section(root, id).join('\n')}');
        if (movings.containsKey(id)) {
          await tester.tap(find.text('$id Start'));
          await tester.pump();
          debugPrint('C26TEST ${platform.name} $id after Start\n${section(root, id).join('\n')}');
          for (var i = 0; i < 6; i++) {
            await tester.pump(const Duration(milliseconds: 1200));
            debugPrint('C26TEST ${platform.name} $id step ${i + 1}\n${section(root, id).join('\n')}');
          }
        }
        // Reduced motion: the same fixture with disableAnimations.
        await tester.pumpWidget(const SizedBox());
        handle.dispose();
        debugDefaultTargetPlatformOverride = null;
      });
    }
  }
}
