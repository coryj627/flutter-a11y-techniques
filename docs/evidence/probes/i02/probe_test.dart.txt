String _focus() => FocusManager.instance.primaryFocus?.debugLabel ?? 'other';

Future<List<String>> _walk(
  WidgetTester tester,
  LogicalKeyboardKey key, {
  int n = 60,
  bool shift = false,
}) async {
  final out = <String>[];
  for (var i = 0; i < n; i++) {
    if (shift) await tester.sendKeyDownEvent(LogicalKeyboardKey.shiftLeft);
    await tester.sendKeyEvent(key);
    if (shift) await tester.sendKeyUpEvent(LogicalKeyboardKey.shiftLeft);
    await tester.pumpAndSettle();
    out.add(_focus());
  }
  return out;
}

/// Identifiers of semantics nodes in traversal order (what a screen reader
/// moving forward meets), depth first.
List<String> _readingOrder(WidgetTester tester) {
  final out = <String>[];
  void visit(SemanticsNode node) {
    final d = node.getSemanticsData();
    if (!node.isMergedIntoParent && d.identifier.isNotEmpty) {
      out.add(d.identifier);
    }
    for (final child in node.debugListChildrenInOrder(
      DebugSemanticsDumpOrder.traversalOrder,
    )) {
      visit(child);
    }
  }

  visit(
    tester.binding.renderViews.first.owner!.semanticsOwner!.rootSemanticsNode!,
  );
  return out;
}

String _pressedFocus(WidgetTester tester) {
  final f = FocusManager.instance.primaryFocus;
  final ctx = f?.context;
  if (ctx == null) return 'none';
  final text = find.descendant(of: find.byElementPredicate((e) => e == ctx), matching: find.byType(Text));
  if (text.evaluate().isNotEmpty) return (tester.widget<Text>(text.first).data ?? '?');
  return 'other(${f.runtimeType})';
}
