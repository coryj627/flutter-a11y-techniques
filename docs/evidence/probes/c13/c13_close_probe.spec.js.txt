// C13 probe round 2 (throwaway, not committed; kept as
// docs/evidence/probes/c13/c13_close_probe.spec.js.txt): what the page and
// Chromium's accessibility tree give the SnackBar's Close button, Undo, and
// the Back button (an IconButton with a tooltip, like Close), in Chromium
// and WebKit, once good-action's message is shown.
import { test } from '@playwright/test';

const log = (...a) => console.log('C13CLOSE', ...a.map((x) => String(x).replace(/\n/g, '\\n')));

test('Close, Undo, and Back', async ({ page, browserName }) => {
  await page.goto('/#/t/C13/good-action');
  const show = page.locator('[flt-semantics-identifier="c13-good-action-show"]');
  await show.waitFor();
  await page.waitForTimeout(800);
  await show.click();
  await page.waitForTimeout(1500);
  for (const name of ['Close', 'Undo', 'Back']) {
    const el = page.getByRole('button', { name, exact: true });
    const html = await el.evaluate((e) => e.outerHTML.replace(/ style="[^"]*"/g, '').replace(/ id="flt-semantic-node-\d+"/g, ''));
    log(browserName, name, 'element:', html);
    const desc = await el.evaluate((e) => ({
      ariaLabel: e.getAttribute('aria-label'),
      ariaDescription: e.getAttribute('aria-description'),
      ariaDescribedby: e.getAttribute('aria-describedby'),
      title: e.getAttribute('title'),
      text: e.textContent,
    }));
    log(browserName, name, 'attributes:', JSON.stringify(desc));
  }
  // Then the pointer on Close, which shows its tooltip (IconButton.tooltip),
  // and keyboard focus on it (Tab from Undo), each read the same way.
  const close = page.getByRole('button', { name: 'Close', exact: true });
  const box = await close.boundingBox();
  await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2);
  await page.waitForTimeout(1500);
  const hovered = await page.evaluate(() => [...document.querySelectorAll('flt-semantics[role="button"]')]
    .map((e) => `${e.getAttribute('aria-label') ?? ''}|${e.textContent}|owns=${e.getAttribute('aria-owns') ?? ''}`));
  log(browserName, 'buttons with the pointer on Close:', JSON.stringify(hovered));
  const tipText = await page.evaluate(() => [...document.querySelectorAll('flt-semantics')]
    .filter((e) => !e.querySelector('flt-semantics') && e.textContent.trim() === 'Close').length);
  log(browserName, 'leaf elements reading "Close" with the pointer on it:', tipText);
  if (browserName === 'chromium') {
    const cdp2 = await page.context().newCDPSession(page);
    const { nodes } = await cdp2.send('Accessibility.getFullAXTree');
    for (const n of nodes) {
      const name = n.name?.value ?? '';
      if (!name.includes('Close')) continue;
      log(browserName, 'AX with the pointer on Close', `role=${n.role?.value}`, `name="${name}"`, `description="${n.description?.value ?? ''}"`);
    }
  }
  await page.mouse.move(1, 1);
  await page.waitForTimeout(2500);
  await page.getByRole('button', { name: 'Undo', exact: true }).focus();
  await page.keyboard.press('Tab');
  await page.waitForTimeout(1500);
  const focused = await page.evaluate(() => `${document.activeElement?.getAttribute('role')} ${document.activeElement?.textContent}`);
  const leaves = await page.evaluate(() => [...document.querySelectorAll('flt-semantics')]
    .filter((e) => !e.querySelector('flt-semantics') && e.textContent.trim() === 'Close').length);
  log(browserName, 'keyboard focus on', focused, '; leaf elements reading "Close":', leaves);
  if (browserName !== 'chromium') return;
  const cdp = await page.context().newCDPSession(page);
  const { nodes } = await cdp.send('Accessibility.getFullAXTree');
  for (const n of nodes) {
    const name = n.name?.value ?? '';
    if (!['Close', 'Undo', 'Back'].includes(name)) continue;
    const props = (n.properties ?? []).map((p) => `${p.name}=${JSON.stringify(p.value?.value)}`).join(' ');
    log(browserName, 'AX', `role=${n.role?.value}`, `name="${name}"`, `description="${n.description?.value ?? ''}"`, props);
  }
});
