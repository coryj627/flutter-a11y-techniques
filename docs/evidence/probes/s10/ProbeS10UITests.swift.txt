// S10 probe (throwaway, not committed; kept as
// docs/evidence/probes/s10/ProbeS10UITests.swift.txt): for each fixture of
// lib/probe.dart, opened by URL: the XCUITest tree; for the fixtures with a
// control, a tap on each element that is a button and, for p13, a tap where
// the hidden Reviews button is drawn; one to three Tabs then Enter, from a
// fresh launch each; and with VoiceOver on, a walk from the fixture's
// "Before" text to its taps line.
import XCTest

final class ProbeS10UITests: XCTestCase {
  let app = XCUIApplication()
  let ids = (1...25).map { String(format: "p%02d", $0) }
  let withControls = ["p13", "p14", "p15", "p22", "p23"]

  func log(_ s: String) { print("S10PROBE \(s)") }

  func tapsElement(_ id: String) -> XCUIElement { app.descendants(matching: .any)["\(id)-taps"] }

  func open(_ id: String) {
    for _ in 1...3 {
      app.terminate()
      app.open(URL(string: "fluttera11y:///\(id)")!)
      if tapsElement(id).waitForExistence(timeout: 30) { sleep(1); return }
    }
    XCTFail("\(id) did not open")
  }

  func taps(_ id: String) -> String {
    sleep(2)
    return tapsElement(id).exists ? tapsElement(id).label : "(gone)"
  }

  func dump(_ id: String) {
    log("==== \(id)")
    for e in app.descendants(matching: .any).allElementsBoundByIndex {
      let t = e.elementType
      if t == .window || t == .application { continue }
      if t == .other && e.label.isEmpty && e.identifier.isEmpty { continue }
      log("\(id) EL type=\(t.rawValue) id=\(e.identifier) label=\(e.label.debugDescription) value=\(String(describing: e.value)) frame=\(e.frame)")
    }
  }

  func testTree() {
    for id in ids {
      open(id)
      dump(id)
      guard withControls.contains(id) else { continue }
      let buttons = app.buttons.allElementsBoundByIndex.map { $0.label }.filter { $0 != "Back" && !$0.isEmpty }
      for label in buttons {
        open(id)
        let b = app.buttons[label]
        if !b.exists { log("\(id) button \(label.debugDescription) not found again"); continue }
        b.tap()
        log("\(id) tap on button \(label.debugDescription): \(taps(id))")
        if id == "p22" || id == "p23" { dump("\(id) after tap") }
      }
      if id == "p13" || id == "p14" {
        // The replaced node; a tap at its right part, where p13's Reviews
        // button is drawn just after the stars, at several points.
        open(id)
        let node = app.descendants(matching: .any).matching(NSPredicate(format: "label BEGINSWITH 'Rating:'")).firstMatch
        log("\(id) replaced node frame=\(node.frame) exists=\(node.exists)")
        for x in stride(from: 0.30, through: 0.60, by: 0.05) {
          open(id)
          node.coordinate(withNormalizedOffset: CGVector(dx: x, dy: 0.5)).tap()
          log("\(id) tap at \(x) of the replaced node: \(taps(id))")
        }
      }
    }
  }

  func testKeys() {
    for id in withControls {
      for tabs in 1...3 {
        open(id)
        for _ in 0..<tabs {
          app.typeKey(XCUIKeyboardKey.tab.rawValue, modifierFlags: [])
          usleep(900_000)
        }
        app.typeKey("\n", modifierFlags: [])
        log("\(id) \(tabs) Tab(s) then Enter: \(taps(id))")
      }
    }
  }

  func testVoiceOver() throws {
    #if compiler(>=6.4)
    guard #available(iOS 27.0, *) else { throw XCTSkip("needs iOS 27") }
    let vo = XCUIDevice.shared.voiceOverService
    func norm(_ s: String) -> String { s.split(whereSeparator: { $0.isWhitespace }).joined(separator: " ") }
    func say() -> String {
      do { return norm(try vo.moveForward().utterance) } catch { return "(error \((error as NSError).code))" }
    }
    for id in ids {
      open(id)
      try vo.enable()
      var walk: [String] = []
      for _ in 0..<24 {
        let s = say()
        walk.append(s)
        if s.hasPrefix("\(id) taps") { break }
      }
      log("VO \(id) walk: \(walk.joined(separator: " / "))")
      try vo.disable()
    }
    #else
    throw XCTSkip("needs Xcode 27")
    #endif
  }
}
