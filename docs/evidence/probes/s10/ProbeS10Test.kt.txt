// S10 probe (throwaway, not committed; kept as
// docs/evidence/probes/s10/ProbeS10Test.kt.txt). Runs against lib/probe.dart
// (Gradle -Ptarget=lib/probe.dart). probe() dumps the nodes of each fixture,
// the nodes between its "Before" text and its taps line in the tree's
// order, then performs ACTION_CLICK on each clickable node among them and
// reads the taps line (p22 and p23: and the replacement's name after);
// keys() tabs through the fixtures with a control with `input keyevent`,
// logs the deepest node with input focus after each Tab, and presses Enter,
// then Space, after one to three Tabs, each from a fresh launch;
// talkBackWalk() moves TalkBack focus to each of the fixture's nodes in
// turn.
package dev.fluttera11y.flutter_a11y_techniques

import android.content.Intent
import android.graphics.Rect
import android.util.Log
import android.view.accessibility.AccessibilityNodeInfo
import android.view.accessibility.AccessibilityWindowInfo
import androidx.test.ext.junit.runners.AndroidJUnit4
import org.junit.Test
import org.junit.runner.RunWith

@RunWith(AndroidJUnit4::class)
class ProbeS10Test : TechniqueTest("S10") {
    private val tag = "S10PROBE"
    private val ids = (1..25).map { "p%02d".format(it) }
    private val withControls = listOf("p13", "p14", "p15", "p22", "p23")

    private fun launch(fixture: String) {
        val intent = Intent(inst.targetContext, MainActivity::class.java)
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK)
            .putExtra("route", "/$fixture")
        inst.targetContext.startActivity(intent)
        val end = System.currentTimeMillis() + 20_000
        while (!anyWindowHas { it.viewIdResourceName == "$fixture-taps" }) {
            if (System.currentTimeMillis() > end) throw AssertionError("$fixture did not open")
            Thread.sleep(250)
        }
        device.waitForIdle()
        Thread.sleep(1_000)
    }

    private fun describe(n: AccessibilityNodeInfo): String {
        val r = Rect().also { n.getBoundsInScreen(it) }
        val actions = n.actionList.joinToString(",") { a ->
            val name = when (a.id) {
                AccessibilityNodeInfo.ACTION_CLICK -> "click"
                AccessibilityNodeInfo.ACTION_FOCUS -> "focus"
                AccessibilityNodeInfo.ACTION_ACCESSIBILITY_FOCUS -> "a11yfocus"
                AccessibilityNodeInfo.ACTION_CLEAR_ACCESSIBILITY_FOCUS -> "clra11y"
                AccessibilityNodeInfo.ACTION_LONG_CLICK -> "longclick"
                AccessibilityNodeInfo.ACTION_SCROLL_FORWARD -> "scrollfwd"
                AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD -> "scrollback"
                else -> a.id.toString()
            }
            if (a.label != null) "$name(${a.label})" else name
        }
        val range = n.rangeInfo?.let { " range=${it.type}:${it.min}..${it.max}@${it.current}" } ?: ""
        return "${n.viewIdResourceName ?: "-"} | ${n.type()} | desc=\"${n.name().replace("\n", "\\n")}\" " +
            "text=\"${(n.text ?: "").toString().replace("\n", "\\n")}\" state=\"${n.stateDescription ?: ""}\" " +
            "hint=\"${n.hintText ?: ""}\" heading=${n.isHeading} clickable=${n.isClickable} " +
            "focusable=${n.isFocusable} screenReaderFocusable=${n.isScreenReaderFocusable} " +
            "importantForA11y=${n.isImportantForAccessibility} $r [$actions] children=${n.childCount}$range"
    }

    private fun collect(root: AccessibilityNodeInfo?, depth: Int, out: MutableList<Pair<Int, AccessibilityNodeInfo>>) {
        if (root == null) return
        out.add(depth to root)
        for (i in 0 until root.childCount) collect(root.getChild(i), depth + 1, out)
    }

    private fun appNodes(): List<Pair<Int, AccessibilityNodeInfo>> {
        val out = mutableListOf<Pair<Int, AccessibilityNodeInfo>>()
        for (w in automation.windows) if (w.type == AccessibilityWindowInfo.TYPE_APPLICATION) collect(w.root, 0, out)
        return out
    }

    private fun label(n: AccessibilityNodeInfo) = n.name().ifEmpty { n.text?.toString().orEmpty() }
        .ifEmpty { n.tooltipText?.toString().orEmpty() }

    private fun taps(id: String): String =
        appNodes().map { it.second }.firstOrNull { it.viewIdResourceName == "$id-taps" }?.let { label(it) } ?: "(gone)"

    /** The nodes after the fixture's "Before" text and before its taps line, in the tree's order. */
    private fun own(id: String): List<AccessibilityNodeInfo> {
        val all = appNodes().map { it.second }
        val start = all.indexOfFirst { label(it) == "Before $id" }
        val end = all.indexOfFirst { it.viewIdResourceName == "$id-taps" }
        if (start < 0 || end < 0 || end <= start) return emptyList()
        return all.subList(start + 1, end)
    }

    /** [own] once it holds a node that [match] accepts, waiting up to ten seconds. */
    private fun ownWhen(id: String, match: (AccessibilityNodeInfo) -> Boolean): AccessibilityNodeInfo? {
        val end = System.currentTimeMillis() + 10_000
        while (System.currentTimeMillis() < end) {
            own(id).firstOrNull(match)?.let { return it }
            Thread.sleep(250)
        }
        return null
    }

    /** Round 2: the clicks and the custom action, each from a fresh launch, waiting for the node. */
    @Test
    fun clicks() {
        for (id in withControls) {
            launch(id)
            val clickables = own(id).filter { it.isClickable }.map { label(it) }
            for (name in clickables) {
                launch(id)
                val node = ownWhen(id) { label(it) == name && it.isClickable }
                if (node == null) {
                    Log.i(tag, "$id \"$name\": not found again")
                    continue
                }
                val ok = node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
                Thread.sleep(2_000)
                Log.i(tag, "$id ACTION_CLICK on \"${name.replace("\n", "\\n")}\" returned $ok; ${taps(id)}")
                if (id == "p22" || id == "p23") {
                    for (n in own(id)) Log.i(tag, "$id after click ${describe(n)}")
                }
            }
        }
        launch("p14")
        val node = ownWhen("p14") { label(it).startsWith("Rating") }
        val custom = node?.actionList?.firstOrNull { it.label?.toString() == "Read reviews" }
        if (node != null && custom != null) {
            val ok = node.performAction(custom.id)
            Thread.sleep(2_000)
            Log.i(tag, "p14 custom action \"Read reviews\" (${custom.id}) returned $ok; ${taps("p14")}")
        } else {
            Log.i(tag, "p14 custom action not found: node=${node != null}")
        }
    }

    @Test
    fun probe() {
        for (id in ids) {
            launch(id)
            Log.i(tag, "==== $id")
            for ((depth, n) in appNodes()) Log.i(tag, "$id all ${"  ".repeat(depth)}${describe(n)}")
            for (n in own(id)) Log.i(tag, "$id own ${describe(n)}")
            val clickables = own(id).filter { it.isClickable }.map { label(it) }
            for (name in clickables) {
                launch(id)
                val node = own(id).firstOrNull { label(it) == name && it.isClickable }
                if (node == null) {
                    Log.i(tag, "$id \"$name\": not found again")
                    continue
                }
                val ok = node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
                Thread.sleep(2_000)
                Log.i(tag, "$id ACTION_CLICK on \"${name.replace("\n", "\\n")}\" returned $ok; ${taps(id)}")
                if (id == "p22" || id == "p23") {
                    for (n in own(id)) Log.i(tag, "$id after click ${describe(n)}")
                }
            }
            if (id == "p14") {
                launch(id)
                val node = own(id).firstOrNull { label(it).startsWith("Rating") }
                val custom = node?.actionList?.firstOrNull { it.label?.toString() == "Read reviews" }
                if (node != null && custom != null) {
                    val ok = node.performAction(custom.id)
                    Thread.sleep(2_000)
                    Log.i(tag, "p14 custom action \"Read reviews\" (${custom.id}) returned $ok; ${taps(id)}")
                } else {
                    Log.i(tag, "p14 custom action not found")
                }
            }
        }
    }

    private fun key(name: String) {
        device.executeShellCommand("input keyevent KEYCODE_$name")
        Thread.sleep(600)
    }

    private fun focusedLabel(): String {
        val f = appNodes().map { it.second }.lastOrNull { it.isFocused } ?: return "(none)"
        return label(f).replace("\n", "\\n").ifEmpty { "(${f.type()} no name)" }
    }

    @Test
    fun keys() {
        for (id in withControls) {
            launch(id)
            val stops = mutableListOf<String>()
            for (i in 0 until 5) {
                key("TAB")
                stops.add(focusedLabel())
            }
            Log.i(tag, "$id Tab stops: ${stops.joinToString(" -> ")}")
            for (k in listOf("ENTER", "SPACE")) {
                for (n in 1..3) {
                    launch(id)
                    for (i in 0 until n) key("TAB")
                    val on = focusedLabel()
                    key(k)
                    Thread.sleep(1_500)
                    Log.i(tag, "$id $n Tab(s) (on $on) then $k: ${taps(id)}")
                }
            }
        }
    }

    @Test
    fun talkBackWalk() {
        assumeTalkBack()
        for (id in ids) {
            launch(id)
            Thread.sleep(6_000)
            val parts = own(id).filter { label(it).isNotEmpty() || it.isClickable || it.rangeInfo != null }
            parts.forEachIndexed { i, n ->
                val marker = "s10-$id-$i"
                Log.i(tag, "$marker is \"${label(n).replace("\n", "\\n")}\" ${n.type()}")
                Log.i(talkBackTag, "FOCUS $marker")
                try {
                    talkBackFocus(marker) { n }
                } catch (e: AssertionError) {
                    Log.i(tag, "$marker: TalkBack focus refused")
                }
                Thread.sleep(6_000)
            }
            Log.i(talkBackTag, "END $id")
        }
    }
}
