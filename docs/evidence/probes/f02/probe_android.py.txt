"""F02 probe on Android: dumps the uiautomator tree of each probe page, and
again after typing into some fields. Run under tool/with_devices.py."""
import re
import subprocess
import sys
import time
import xml.etree.ElementTree as ET

APK = sys.argv[1]
OUT = sys.argv[2]
PKG = "dev.fluttera11y.flutter_a11y_techniques"


def adb(*args, check=True):
    return subprocess.run(["adb", *args], capture_output=True, text=True, check=check).stdout


def dump():
    for _ in range(3):
        adb("shell", "uiautomator", "dump", "/sdcard/probe.xml", check=False)
        xml = adb("shell", "cat", "/sdcard/probe.xml", check=False)
        if xml.strip().startswith("<?xml"):
            return xml
        time.sleep(2)
    raise SystemExit("no dump")


def nodes(xml):
    root = ET.fromstring(xml)
    for n in root.iter("node"):
        yield n


def describe(xml):
    lines = []
    for n in nodes(xml):
        a = n.attrib
        if a.get("package") != PKG:
            continue
        keep = {k: a.get(k, "") for k in ("resource-id", "class", "text", "content-desc", "hint",
                                          "clickable", "focusable", "password", "bounds")}
        if not (keep["resource-id"] or keep["text"] or keep["content-desc"] or keep["hint"]):
            continue
        lines.append(" ".join(f"{k}={v!r}" for k, v in keep.items() if v not in ("", "false")))
    return "\n".join(lines)


def centre(xml, rid):
    for n in nodes(xml):
        if n.attrib.get("resource-id") == rid:
            x1, y1, x2, y2 = map(int, re.findall(r"\d+", n.attrib["bounds"]))
            return (x1 + x2) // 2, (y1 + y2) // 2
    raise SystemExit(f"no {rid}")


def by_text(xml, text):
    for n in nodes(xml):
        if n.attrib.get("content-desc") == text or n.attrib.get("text") == text:
            x1, y1, x2, y2 = map(int, re.findall(r"\d+", n.attrib["bounds"]))
            return (x1 + x2) // 2, (y1 + y2) // 2
    raise SystemExit(f"no {text}")


def tap(xy):
    adb("shell", "input", "tap", str(xy[0]), str(xy[1]))
    time.sleep(1.5)


out = []
adb("install", "-r", APK)
adb("shell", "am", "force-stop", PKG)
adb("shell", "monkey", "-p", PKG, "1")
time.sleep(8)
x = dump()
out.append("=== page 1, nothing typed\n" + describe(x))
tap(centre(x, "i05"))
x = dump()
out.append("=== page 1, i05 focused, empty\n" + describe(x))
tap(centre(x, "i03"))
adb("shell", "input", "text", "abc")
time.sleep(1.5)
tap(centre(x, "i01"))
adb("shell", "input", "text", "abc")
time.sleep(1.5)
adb("shell", "input", "keyevent", "4")  # put the keyboard away
time.sleep(1.5)
x = dump()
out.append("=== page 1, abc typed into i03 and i01, keyboard away\n" + describe(x))
tap(by_text(x, "Page 2"))
time.sleep(1)
x = dump()
out.append("=== page 2, nothing typed\n" + describe(x))
tap(centre(x, "i10"))
adb("shell", "input", "text", "abc")
time.sleep(1.5)
x = dump()
out.append("=== page 2, abc typed into i10, still editing\n" + describe(x))
adb("shell", "input", "keyevent", "4")
time.sleep(1.5)
x = dump()
out.append("=== page 2, keyboard away\n" + describe(x))
adb("shell", "am", "force-stop", PKG)
open(OUT, "w").write("\n\n".join(out) + "\n")
print("\n\n".join(out))
