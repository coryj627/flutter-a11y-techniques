// C18 probe (throwaway, not committed; kept as
// docs/evidence/probes/c18/probe.dart.txt): what cards built from the SDK
// expose, and what reaches and activates their parts. One fixture per
// route, `/p01` to `/p20`; `/` lists them all; `/dest` is an in-app
// destination that the fixtures with `linkUrl: '#/dest'` point to.
//
// Each fixture is one card (or a list of cards) between a "Before pNN" text
// and a "pNN taps: whole=N a=N b=N" text that counts the taps that reached
// each handler: the card as a whole, its first action, its second action.
// Off the web the card-as-a-link fixtures also open `/dest`; on the web the
// browser follows the address itself.
import 'package:flutter/cupertino.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';
import 'package:flutter/services.dart';

import 'techniques/s02_informative_images/example_image.dart';

void main() {
  runApp(const ProbeApp());
  SemanticsBinding.instance.ensureSemantics();
}

final dest = Uri.parse('#/dest');
int destOpened = 0;

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      onGenerateRoute: (settings) {
        final name = settings.name ?? '/';
        Widget page;
        if (name == '/dest') {
          destOpened++;
          debugPrint('C18PROBE DEST $destOpened');
          page = Scaffold(
            appBar: AppBar(title: const Text('Destination')),
            body: Text('Arrived $destOpened'),
          );
        } else if (fixtures.containsKey(name.replaceFirst('/', ''))) {
          page = FixtureScreen(ids: [name.replaceFirst('/', '')]);
        } else {
          page = FixtureScreen(ids: fixtures.keys.toList());
        }
        return MaterialPageRoute<void>(settings: settings, builder: (_) => page);
      },
    );
  }
}

/// What a fixture gets: its id, the context, and the handler of each part
/// (0 the whole card, 1 the first action, 2 the second action).
typedef Build = Widget Function(String id, BuildContext context, void Function(int) tap);

Widget heading(String id, {bool level = true}) => level
    ? Semantics(headingLevel: 3, child: Text('$id Card heading', style: const TextStyle(fontSize: 20)))
    : Text('$id Card heading', style: const TextStyle(fontSize: 20));

Widget body(String id) => Text('$id Body text of the card. A second sentence.');

Widget picture(String id, {bool exclude = false}) => Image.memory(
  sunrisePng,
  width: 96,
  height: 64,
  excludeFromSemantics: exclude,
  semanticLabel: exclude ? null : '$id picture',
);

Widget pad(Widget child) => Padding(padding: const EdgeInsets.all(12), child: child);

Widget column(List<Widget> children) => Column(
  crossAxisAlignment: CrossAxisAlignment.start,
  mainAxisSize: MainAxisSize.min,
  spacing: 8,
  children: children,
);

/// C03's `BrowserKeys`: on the web Enter and Space go to the browser.
Widget browserKeys(Widget child) => kIsWeb
    ? Shortcuts(
        shortcuts: const {
          SingleActivator(LogicalKeyboardKey.enter): DoNothingAndStopPropagationIntent(),
          SingleActivator(LogicalKeyboardKey.space): DoNothingAndStopPropagationIntent(),
        },
        child: child,
      )
    : child;

Widget partsCard(String id, void Function(int) tap, {required bool container, String a = 'Read about cards', String b = 'Save'}) => Card(
  semanticContainer: container,
  child: pad(
    column([
      heading(id),
      body(id),
      picture(id),
      Row(
        children: [
          TextButton(onPressed: () => tap(1), child: Text('$id $a')),
          TextButton(onPressed: () => tap(2), child: Text('$id $b')),
        ],
      ),
    ]),
  ),
);

Widget inkCard(String id, void Function(int) tap, {required bool container, List<Widget> extra = const []}) => Card(
  semanticContainer: container,
  clipBehavior: Clip.antiAlias,
  child: InkWell(
    onTap: () => tap(0),
    child: pad(column([heading(id), body(id), picture(id, exclude: true), ...extra])),
  ),
);

Widget smallCard(String id, int n, void Function(int) tap) => Card(
  semanticContainer: false,
  child: pad(
    column([
      Semantics(headingLevel: 3, child: Text('$id Card $n heading')),
      Text('$id Card $n body'),
      TextButton(onPressed: () => tap(n), child: Text('$id Save $n')),
    ]),
  ),
);

class FixtureScreen extends StatefulWidget {
  const FixtureScreen({super.key, required this.ids});

  final List<String> ids;

  @override
  State<FixtureScreen> createState() => _FixtureScreenState();
}

class _FixtureScreenState extends State<FixtureScreen> {
  final taps = <String, List<int>>{};

  static const linkFixtures = {'p09', 'p11'};

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.ids.length == 1 ? widget.ids.single : 'C18 probe')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          spacing: 12,
          children: [
            for (final id in widget.ids) ...[
              Text('Before $id'),
              Builder(
                builder: (context) => fixtures[id]!.$2(id, context, (which) {
                  debugPrint('C18PROBE TAP $id $which');
                  setState(() => (taps[id] ??= [0, 0, 0, 0])[which]++);
                  if (!kIsWeb && linkFixtures.contains(id)) {
                    Navigator.of(context).pushNamed('/dest');
                  }
                }),
              ),
              Semantics(
                identifier: '$id-taps',
                child: Text(
                  '$id taps: whole=${taps[id]?[0] ?? 0} '
                  'a=${taps[id]?[1] ?? 0} b=${taps[id]?[2] ?? 0} c=${taps[id]?[3] ?? 0}',
                ),
              ),
            ],
          ],
        ),
      ),
    );
  }
}
