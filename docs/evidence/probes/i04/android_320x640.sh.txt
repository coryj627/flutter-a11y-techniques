#!/bin/sh
# I04 Android suite on a 320 by 640, 160 dpi screen, as on CI. Run from the
# worktree root under tool/with_devices.py. The size and density are reset
# whatever happens.
reset() {
  adb shell wm size reset
  adb shell wm density reset
  echo "== reset: $(adb shell wm size | tr '\n' ' ') $(adb shell wm density | tr '\n' ' ')"
}
trap reset EXIT INT TERM
adb shell wm size 320x640
adb shell wm density 160
echo "== set: $(adb shell wm size | tr '\n' ' ') $(adb shell wm density | tr '\n' ' ')"
cd android || exit 1
./gradlew --no-daemon -Pkotlin.compiler.execution.strategy=in-process app:connectedDebugAndroidTest \
  -Pandroid.testInstrumentationRunnerArguments.class=dev.fluttera11y.flutter_a11y_techniques.I04KeyboardActionsTest \
  -Pandroid.testInstrumentationRunnerArguments.timeout_msec=180000 2>&1 \
  | grep -E "FAILED|Failure|expected|Tests on|tests completed|BUILD|AssertionError"
