import { test } from '@playwright/test';

const status = async (page, id) =>
  ((await page.locator(`[flt-semantics-identifier="${id}"]`).first().textContent()) ?? '').trim();
const focus = async (page) => (await status(page, 'focus')).replace('Focus: ', '');
const say = (...parts) => console.log('I04PROBE', ...parts);

async function snapshot(page, what) {
  const active = await page.evaluate(() => {
    const e = document.activeElement;
    return e ? `${e.tagName.toLowerCase()}${e.getAttribute('flt-semantics-identifier') ? '#' + e.getAttribute('flt-semantics-identifier') : ''}` : 'null';
  });
  say(`${what} | focus=${await focus(page)} | active=${active} | ${await status(page, 'log')} | ${await status(page, 'texts')} | ${await status(page, 'scroll')} | ${await status(page, 'keys')}`);
}

async function open(page) {
  await page.goto('about:blank');
  await page.goto('/#/probe');
  await page.locator('[flt-semantics-identifier="focus"]').waitFor();
  await page.waitForTimeout(800);
}

async function tabTo(page, label, max = 60) {
  for (let i = 0; i < max; i++) {
    if ((await focus(page)) === label) return true;
    await page.keyboard.press('Tab');
    await page.waitForTimeout(80);
  }
  return (await focus(page)) === label;
}

async function press(page, key, wait = 250) {
  await page.keyboard.press(key);
  await page.waitForTimeout(wait);
}

test('tab path', async ({ page }) => {
  await open(page);
  const path = [];
  for (let i = 0; i < 40; i++) {
    await press(page, 'Tab', 100);
    path.push(await focus(page));
  }
  say('tab path:', path.join(' '));
});

test('q01 and q02: letter shortcuts around a field', async ({ page }) => {
  for (const id of ['q01', 'q02']) {
    await open(page);
    say(id, 'reached', await tabTo(page, id));
    await page.keyboard.type('afg b', { delay: 60 });
    await page.waitForTimeout(300);
    await snapshot(page, `${id} typed "afg b" in the field`);
    say(id, 'reached button', await tabTo(page, `${id}-btn`));
    await press(page, 'f');
    await press(page, 'g');
    await press(page, 'Shift+F');
    await snapshot(page, `${id} f, g, Shift+F on the button`);
  }
});

test('q05 and q06: viewers', async ({ page }) => {
  for (const id of ['q05', 'q06']) {
    await open(page);
    say(id, 'reached', await tabTo(page, id));
    await snapshot(page, `${id} reached`);
    for (const key of ['ArrowRight', 'ArrowRight', 'ArrowLeft', 'End', 'Home', 'f', 'ArrowDown', 'ArrowUp', 'PageDown']) {
      await press(page, key);
      await snapshot(page, `${id} ${key}`);
    }
    await page.keyboard.down('ArrowRight');
    await page.waitForTimeout(1500);
    await page.keyboard.up('ArrowRight');
    await page.waitForTimeout(300);
    await snapshot(page, `${id} ArrowRight held 1.5 s`);
  }
});

test('q08: carousel items', async ({ page }) => {
  await open(page);
  await tabTo(page, 'q06');
  await press(page, 'Tab');
  const role = await page.evaluate(() => {
    const e = document.activeElement;
    return e ? `${e.tagName} role=${e.getAttribute('role')} name=${e.getAttribute('aria-label') ?? e.textContent}` : 'null';
  });
  say('q08 first carousel stop element:', role);
  for (const key of ['Enter', 'Space', 'ArrowRight', 'ArrowLeft', 'Tab', 'Enter']) {
    await press(page, key, 400);
    await snapshot(page, `q08 ${key}`);
  }
});
