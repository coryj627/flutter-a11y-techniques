// S09 probe (throwaway, not committed; kept as
// docs/evidence/probes/s09/probe.dart.txt): what the SDK's grouping
// mechanisms expose, and what reaches and activates the controls inside a
// group. One fixture per route, `/s01` to `/s18`; `/` lists them all.
//
// Each fixture sits between a "Before sNN" text and a "sNN taps: on=N off=N
// info=N" text that counts the taps that reached each handler.
import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';

void main() {
  runApp(const ProbeApp());
  SemanticsBinding.instance.ensureSemantics();
}

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      onGenerateRoute: (settings) {
        final name = (settings.name ?? '/').replaceFirst('/', '');
        final page = fixtures.containsKey(name)
            ? FixtureScreen(ids: [name])
            : FixtureScreen(ids: fixtures.keys.toList());
        return MaterialPageRoute<void>(settings: settings, builder: (_) => page);
      },
    );
  }
}

/// What a fixture gets: its id and the handler of each control
/// ("on", "off", "info").
typedef Build = Widget Function(String id, void Function(String) tap);

Widget pair(String id, String label, String value) => Column(
  crossAxisAlignment: CrossAxisAlignment.start,
  children: [Text('$id $label'), Text(value, style: const TextStyle(fontSize: 20))],
);

Widget question(String id) =>
    Text('$id Kitchen light', style: const TextStyle(fontSize: 18));

Widget onOff(String id, void Function(String) tap) => Row(
  spacing: 8,
  children: [
    OutlinedButton(onPressed: () => tap('on'), child: Text('$id On')),
    OutlinedButton(onPressed: () => tap('off'), child: Text('$id Off')),
  ],
);

class FixtureScreen extends StatelessWidget {
  const FixtureScreen({super.key, required this.ids});

  final List<String> ids;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('S09 probe')),
      // As the app's PageScaffold: a SingleChildScrollView round a Column,
      // not a ListView, whose items would each be a node of their own.
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(12),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [for (final id in ids) Fixture(id: id)],
        ),
      ),
    );
  }
}

class Fixture extends StatefulWidget {
  const Fixture({super.key, required this.id});

  final String id;

  @override
  State<Fixture> createState() => _FixtureState();
}

class _FixtureState extends State<Fixture> {
  final _taps = <String, int>{};

  @override
  Widget build(BuildContext context) {
    final id = widget.id;
    final (_, build) = fixtures[id]!;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      spacing: 12,
      children: [
        Text('Before $id'),
        build(id, (k) {
          setState(() => _taps[k] = (_taps[k] ?? 0) + 1);
          debugPrint('S09PROBE TAP $id $k');
        }),
        Semantics(
          identifier: '$id-taps',
          container: true,
          child: Text(
            '$id taps: on=${_taps['on'] ?? 0} off=${_taps['off'] ?? 0} info=${_taps['info'] ?? 0}',
          ),
        ),
        const SizedBox(height: 12),
      ],
    );
  }
}
