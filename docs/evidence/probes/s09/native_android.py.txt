"""S09 native comparison runner for Android (throwaway; kept as
docs/evidence/probes/s09/native_android.py.txt). Run under the device lock:

  python3 tool/with_devices.py --label "probe S09 native android" -- python3 <this> <repo>

Installs the app's debug build (with ProbeS09NativeActivity declared in the
debug manifest for the run) and its test APK, built beforehand with
`flutter build apk --debug` and `./gradlew app:assembleDebugAndroidTest`,
then runs ProbeS09NativeTest with `am instrument`:

- nativeTree: without TalkBack; the S09PROBE lines go in
  android-native-tree.txt.
- talkBackWalk, talkBackKeys: with TalkBack on through the project's own
  helper (tool/talkback.py: run_with_talkback, which restores the device);
  the speech per marker as parsed by parse_speech goes in
  android-native-talkback.md, the markers and the focus lines in
  android-native-talkback-markers.txt.
"""
import os
import subprocess
import sys
import time

ROOT = sys.argv[1]
OUT = os.path.join(ROOT, "docs", "evidence", "probes", "s09")
SCRATCH = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, os.path.join(ROOT, "tool"))
import talkback  # noqa: E402
from evidence_common import stop_android_tests  # noqa: E402

PKG = "dev.fluttera11y.flutter_a11y_techniques"
APKS = [
    os.path.join(ROOT, "build/app/outputs/apk/debug/app-debug.apk"),
    os.path.join(ROOT, "build/app/outputs/apk/androidTest/debug/app-debug-androidTest.apk"),
]


def instrument(method):
    talkback.adb("logcat", "-c")
    log_path = os.path.join(SCRATCH, f"android-native-{method}-logcat.txt")
    with open(log_path, "w") as log:
        reader = subprocess.Popen([talkback.ADB, "logcat"], stdout=log, stderr=subprocess.DEVNULL)
        try:
            run = subprocess.run(
                [talkback.ADB, "shell", "am", "instrument", "-w", "-e", "class",
                 f"{PKG}.ProbeS09NativeTest#{method}", f"{PKG}.test/androidx.test.runner.AndroidJUnitRunner"],
                capture_output=True, text=True, timeout=60 * 60)
        finally:
            time.sleep(2)
            reader.terminate()
            reader.wait()
    print(method, run.stdout[-1500:])
    stop_android_tests()
    with open(log_path, errors="replace") as f:
        return f.read()


def keep(logs, name, header):
    lines = [header]
    for log in logs:
        lines += [l.split("S09PROBE: ", 1)[-1] for l in log.splitlines() if "S09PROBE" in l]
    with open(os.path.join(OUT, name), "w") as f:
        f.write("\n".join(lines) + "\n")


def main():
    os.makedirs(OUT, exist_ok=True)
    if not stop_android_tests():
        sys.exit("emulator not clean")
    for apk in APKS:
        print(talkback.adb("install", "-r", "-t", apk)[-200:])
    device = (talkback.adb("shell", "getprop", "ro.build.version.release").strip(),
              talkback.adb("shell", "wm", "size").strip().replace("\n", " "),
              talkback.adb("shell", "wm", "density").strip().replace("\n", " "))
    header = f"device: Android {device[0]}; {device[1]}; {device[2]}"
    print(header)
    tb = talkback.adb("shell", "dumpsys", "package", "com.google.android.marvin.talkback")
    version = next((l.strip() for l in tb.splitlines() if "versionName" in l), "versionName unknown")
    header += f"; TalkBack {version}"
    keep([instrument("nativeTree")], "android-native-tree.txt", header)
    talkback.exit_on_termination_signals()
    logs = talkback.run_with_talkback(lambda: [instrument("talkBackWalk"), instrument("talkBackKeys")])
    spoken = {}
    for log in logs:
        spoken.update(talkback.parse_speech(log))
    out = [header, "", "| Marker | TalkBack spoke |", "| --- | --- |"]
    for ident, fragments in spoken.items():
        out.append(f"| `{ident}` | {' / '.join(fragments) or '(nothing)'} |")
    with open(os.path.join(OUT, "android-native-talkback.md"), "w") as f:
        f.write("\n".join(out) + "\n")
    keep(logs, "android-native-talkback-markers.txt", header)
    print("\n".join(out))
    talkback.adb("uninstall", f"{PKG}.test")


if __name__ == "__main__":
    main()
