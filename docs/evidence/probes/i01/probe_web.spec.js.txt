// Throwaway: I01 probe on the web. Not committed.
import { test } from '@playwright/test';

const id = (page, x) => page.locator(`[flt-semantics-identifier="${x}"]`);
const log = async (page) => (await id(page, 'log').textContent()) ?? '';
const focus = async (page) => (await id(page, 'focus').textContent()) ?? '';
const active = (page) => page.evaluate(() => {
  const e = document.activeElement;
  if (!e) return 'none';
  const ident = e.getAttribute('flt-semantics-identifier') ?? e.closest('[flt-semantics-identifier]')?.getAttribute('flt-semantics-identifier');
  return `${e.tagName.toLowerCase()} id=${ident ?? '-'} role=${e.getAttribute('role')} tabindex=${e.getAttribute('tabindex')}`;
});
const ids = Array.from({ length: 14 }, (_, i) => `k${String(i + 1).padStart(2, '0')}`);

async function start(page) {
  await page.goto('about:blank');
  await page.goto('/');
  await id(page, 'k14').waitFor();
  await page.waitForTimeout(800);
}

test('elements', async ({ page }) => {
  await start(page);
  for (const x of ids) {
    const e = id(page, x);
    const info = await e.evaluate((el) => {
      const attrs = {};
      for (const a of el.attributes) if (!['style', 'id'].includes(a.name)) attrs[a.name] = a.value;
      return JSON.stringify(attrs);
    });
    const cdp = await page.context().newCDPSession(page);
    console.log(`${x}: ${info}`);
    await cdp.detach();
  }
  const cdp = await page.context().newCDPSession(page);
  const { nodes } = await cdp.send('Accessibility.getFullAXTree');
  for (const n of nodes) {
    const name = n.name?.value ?? '';
    if (/^k\d\d/.test(name)) {
      const props = Object.fromEntries((n.properties ?? []).map((p) => [p.name, p.value?.value]));
      console.log(`AX ${n.role?.value} name=${JSON.stringify(name)} ${JSON.stringify(props)}`);
    }
  }
});

test('tab order, Enter and Space', async ({ page }) => {
  await start(page);
  for (let i = 1; i <= 16; i++) {
    await page.keyboard.press('Tab');
    await page.waitForTimeout(150);
    const where = await active(page);
    const f = await focus(page);
    const before = await log(page);
    // Never press Enter on Back; the probe has no Back button.
    await page.keyboard.press('Enter');
    await page.waitForTimeout(150);
    const afterEnter = await log(page);
    await page.keyboard.press('Space');
    await page.waitForTimeout(150);
    const afterSpace = await log(page);
    console.log(`tab ${i}: ${where} | ${f} | Enter added "${afterEnter.slice(before.length)}" | Space added "${afterSpace.slice(afterEnter.length)}"`);
  }
});

test('pointer: press and slide off', async ({ page }) => {
  await start(page);
  for (const x of ['k01', 'k06', 'k09', 'k10', 'k04', 'k07']) {
    const box = await id(page, x).boundingBox();
    const before = await log(page);
    await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2);
    await page.mouse.down();
    await page.waitForTimeout(200);
    const down = await log(page);
    await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2 + 600, { steps: 10 });
    await page.mouse.up();
    await page.waitForTimeout(300);
    const after = await log(page);
    console.log(`slide off ${x}: on down "${down.slice(before.length)}" | after "${after.slice(down.length)}"`);
  }
  for (const x of ['k04', 'k07', 'k10', 'k13']) {
    const before = await log(page);
    // A click event on the element, as a screen reader's activation sends.
    await id(page, x).dispatchEvent('click');
    await page.waitForTimeout(300);
    console.log(`click event on ${x}: "${(await log(page)).slice(before.length)}"`);
  }
});

test('pointer: hold', async ({ page }) => {
  await start(page);
  for (const x of ['k09', 'k06', 'k10']) {
    const box = await id(page, x).boundingBox();
    const before = await log(page);
    await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2);
    await page.mouse.down();
    const seen = [];
    let waited = 0;
    for (const ms of [100, 300, 600, 1200]) {
      await page.waitForTimeout(ms - waited);
      waited = ms;
      seen.push(`${ms}ms "${(await log(page)).slice(before.length)}"`);
    }
    await page.mouse.up();
    await page.waitForTimeout(300);
    console.log(`hold ${x}: ${seen.join(' ')} | after up "${(await log(page)).slice(before.length)}"`);
    const b2 = await log(page);
    await page.mouse.down();
    await page.waitForTimeout(1200);
    const held = (await log(page)).slice(b2.length);
    await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2 + 600, { steps: 10 });
    await page.mouse.up();
    await page.waitForTimeout(300);
    console.log(`hold 1.2 s then slide off ${x}: while held "${held}" | after "${(await log(page)).slice(b2.length + held.length)}"`);
  }
});
