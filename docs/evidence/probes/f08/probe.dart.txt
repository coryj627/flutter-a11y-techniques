// F08 probe (throwaway, not committed): radio groups.

import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';

void main() {
  runApp(const MaterialApp(home: ProbeScreen()));
  SemanticsBinding.instance.ensureSemantics();
}

class ProbeScreen extends StatelessWidget {
  const ProbeScreen({super.key});

  Widget _box(String text) => ExcludeSemantics(
    child: SizedBox(width: 180, height: 40, child: Align(alignment: Alignment.centerLeft, child: Text(text))),
  );

  @override
  Widget build(BuildContext context) {
    void no(String? _) {}
    Widget tiles(String p, {bool enabled = true}) => Column(children: [
      RadioListTile<String>(value: 'a', enabled: enabled, title: Text('$p Small')),
      RadioListTile<String>(value: 'b', enabled: enabled, title: Text('$p Large')),
    ]);
    return Scaffold(
      body: SafeArea(
        child: SingleChildScrollView(
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              // 01 RadioGroup with two RadioListTiles, second selected
              RadioGroup<String>(groupValue: 'b', onChanged: no, child: tiles('r01')),
              // 02 plain Semantics(label:) around a RadioGroup
              Semantics(label: 'r02 Size', child: RadioGroup<String>(groupValue: 'b', onChanged: no, child: tiles('r02'))),
              // 03 container + explicitChildNodes + label around a RadioGroup
              Semantics(container: true, explicitChildNodes: true, label: 'r03 Size', child: RadioGroup<String>(groupValue: 'b', onChanged: no, child: tiles('r03'))),
              // 04 the same with role: radioGroup
              Semantics(container: true, explicitChildNodes: true, role: SemanticsRole.radioGroup, label: 'r04 Size', child: RadioGroup<String>(groupValue: 'b', onChanged: no, child: tiles('r04'))),
              // 05 bare Radio beside Text, not associated
              RadioGroup<String>(groupValue: 'b', onChanged: no, child: const Column(children: [
                Row(children: [Radio<String>(value: 'a'), Text('r05 Small')]),
                Row(children: [Radio<String>(value: 'b'), Text('r05 Large')]),
              ])),
              // 06 CupertinoRadio merged with Text
              RadioGroup<String>(groupValue: 'b', onChanged: no, child: const Column(children: [
                MergeSemantics(child: Row(children: [CupertinoRadio<String>(value: 'a'), Text('r06 Small')])),
                MergeSemantics(child: Row(children: [CupertinoRadio<String>(value: 'b'), Text('r06 Large')])),
              ])),
              // 07 custom: checked + inMutuallyExclusiveGroup only
              Semantics(container: true, checked: false, inMutuallyExclusiveGroup: true, enabled: true, label: 'r07 Small', onTap: () {}, child: _box('r07a')),
              Semantics(container: true, checked: true, inMutuallyExclusiveGroup: true, enabled: true, label: 'r07 Large', onTap: () {}, child: _box('r07b')),
              // 08 custom with selected too
              Semantics(container: true, checked: false, selected: false, inMutuallyExclusiveGroup: true, enabled: true, label: 'r08 Small', onTap: () {}, child: _box('r08a')),
              Semantics(container: true, checked: true, selected: true, inMutuallyExclusiveGroup: true, enabled: true, label: 'r08 Large', onTap: () {}, child: _box('r08b')),
              // 09 disabled tiles
              RadioGroup<String>(groupValue: 'b', onChanged: no, child: tiles('r09', enabled: false)),
              // 10 radios faked as buttons
              Semantics(container: true, button: true, label: 'r10 Small', onTap: () {}, child: _box('r10a')),
              // 11 text above a group, not associated
              const Text('r11 Size'),
              RadioGroup<String>(groupValue: 'b', onChanged: no, child: tiles('r11')),
            ],
          ),
        ),
      ),
    );
  }
}
