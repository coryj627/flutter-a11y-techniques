// Throwaway: V14 probe on the web (Chromium, Playwright). Not committed;
// kept as docs/evidence/probes/v14/probe_web.spec.js.txt. For each control
// fixture, light and dark: screenshot (u), Tab until the probe's status says
// focus is in the box, screenshot after 1.2 s (f1) and 0.7 s more (f2), a
// touch on an empty part of the page (e), then a fresh page and a touch on
// the control (t), and a mouse click on the control (mc). Writes PNGs and
// the box's rectangle for analyse.py. Then the obscuring fixtures, and what
// the browser draws for its own focus outline on the focused element.
import fs from 'node:fs';
import { test } from '@playwright/test';

const OUT = process.env.V14_OUT;
const ids = (process.env.V14_FIXTURES ?? 'm1 m2 m3 m4 m5 m6 m7 m8 m9 m10 m11 m12 m13 m14 m15 c1 c2 c3 c4 c5 w1 w2 w3 x1 x2 x3 x4 x5').split(' ');

const byId = (page, id) => page.locator(`[flt-semantics-identifier="${id}"]`);
const status = async (page) => {
  const s = byId(page, 'v14-status');
  return (await s.getAttribute('aria-label')) ?? (await s.textContent()) ?? '';
};

async function open(page, route, id) {
  await page.goto('about:blank');
  await page.goto(`/#${route}`);
  await page.waitForFunction((id) => {
    const e = document.querySelector('[flt-semantics-identifier="v14-status"]');
    return e && (e.getAttribute('aria-label') ?? e.textContent ?? '').startsWith(`S ${id} `);
  }, id);
  await page.waitForTimeout(800);
}

test('controls', async ({ page }) => {
  fs.mkdirSync(OUT, { recursive: true });
  await page.setViewportSize({ width: 400, height: 500 });
  for (const id of ids) {
    for (const theme of ['light', 'dark']) {
      const tag = `web-${theme}-${id}`;
      const route = `/c/${id}${theme === 'dark' ? '/dark' : ''}`;
      await open(page, route, id);
      const b = await byId(page, 'v14-box').boundingBox();
      fs.writeFileSync(`${OUT}/${tag}-u.box`, `${b.x} ${b.y} ${b.width} ${b.height} 1\n`);
      const s0 = await status(page);
      await page.screenshot({ path: `${OUT}/${tag}-u.png` });
      for (let i = 0; i < 4 && !(await status(page)).includes('f=1'); i++) {
        await page.keyboard.press('Tab');
        await page.waitForTimeout(400);
      }
      await page.waitForTimeout(1200);
      await page.screenshot({ path: `${OUT}/${tag}-f1.png` });
      const s1 = await status(page);
      const active = await page.evaluate(() => {
        const e = document.activeElement;
        if (!e || e === document.body) return 'body';
        const cs = getComputedStyle(e);
        return `${e.tagName.toLowerCase()} role=${e.getAttribute('role')} outline=${cs.outlineStyle} ${cs.outlineWidth} ${cs.outlineColor} filter=${cs.filter} opacity=${cs.opacity}`;
      });
      await page.waitForTimeout(700);
      await page.screenshot({ path: `${OUT}/${tag}-f2.png` });
      await page.touchscreen.tap(390, 480);
      await page.waitForTimeout(1500);
      await page.screenshot({ path: `${OUT}/${tag}-e.png` });
      const s2 = await status(page);
      await open(page, route, id);
      await page.touchscreen.tap(b.x + b.width / 2, b.y + b.height / 2);
      await page.waitForTimeout(1500);
      await page.screenshot({ path: `${OUT}/${tag}-t.png` });
      const s3 = await status(page);
      await open(page, route, id);
      await page.mouse.click(b.x + b.width / 2, b.y + b.height / 2);
      await page.mouse.move(395, 495);
      await page.waitForTimeout(1500);
      await page.screenshot({ path: `${OUT}/${tag}-mc.png` });
      const s4 = await status(page);
      const text = `u: ${s0}\ntab: ${s1}\ntouch elsewhere: ${s2}\ntap: ${s3}\nmouse click: ${s4}\nactive after Tab: ${active}\n`;
      fs.writeFileSync(`${OUT}/${tag}.status`, text);
      console.log(`V14WEB ${tag} | ${text.replaceAll('\n', ' | ')}`);
    }
  }
});

test('obscure', async ({ page }) => {
  await page.setViewportSize({ width: 400, height: 500 });
  for (const id of ['o1', 'o2', 'o3', 'o4', 'o5']) {
    await page.goto('about:blank');
    await page.goto(`/#/o/${id}`);
    await byId(page, 'v14-status').waitFor();
    await page.waitForTimeout(800);
    console.log(`V14WEB == ${id} start: ${await status(page)}`);
    for (let i = 1; i <= 13; i++) {
      await page.keyboard.press('Tab');
      await page.waitForTimeout(400);
      console.log(`V14WEB   Tab ${i}: ${await status(page)}`);
    }
  }
});
