// V14 probe in a widget test (throwaway, not committed; kept as
// docs/evidence/probes/v14/probe_test.dart.txt). Pumps each fixture of
// lib/probe.dart, as Android and as iOS, light and dark, and writes PNGs of
// the fixture's box: unfocused (u), after Tab (f1), 700 ms later (f2), and
// after a tap on the control in a fresh app (t). tool-free analysis is done
// by analyse.py over the PNGs. Run with V14_OUT set to a folder.
import 'dart:io';
import 'dart:ui' as ui;

import 'package:flutter/foundation.dart';
import 'package:flutter/widgets.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_a11y_techniques/probe.dart';
import 'package:flutter_test/flutter_test.dart';

const controls = [
  'm1', 'm2', 'm3', 'm4', 'm5', 'm6', 'm7', 'm8', 'm9', 'm10', 'm11', 'm12',
  'm13', 'm14', 'm15', 'c1', 'c2', 'c3', 'c4', 'c5', 'w1', 'w2', 'w3', 'x1',
  'x2', 'x3', 'x4', 'x5',
];

final out = Platform.environment['V14_OUT'] ?? '/tmp/v14';

/// The device pixel ratio to render at (round 2 of the measure: 3).
final dpr = double.parse(Platform.environment['V14_DPR'] ?? '1');

Finder get box => find.byWidgetPredicate(
  (w) => w is Semantics && w.properties.identifier == 'v14-box',
);

String status(WidgetTester tester) {
  final t = tester.widget<Text>(
    find.descendant(
      of: find.byWidgetPredicate(
        (w) => w is Semantics && w.properties.identifier == 'v14-status',
      ),
      matching: find.byType(Text),
    ),
  );
  return t.data!;
}

Future<void> save(WidgetTester tester, String name) async {
  final element = box.evaluate().single;
  final render = element.renderObject! as RenderBox;
  // Capture the whole screen, then crop to the box in the analyser: the box
  // is not a repaint boundary of its own.
  final root = tester.binding.renderViews.first;
  final ui.Image image = await (root.debugLayer! as OffsetLayer).toImage(
    Offset.zero & (root.size * dpr),
  );
  final topLeft = render.localToGlobal(Offset.zero);
  await tester.runAsync(() async {
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    File('$out/$name.png')
      ..createSync(recursive: true)
      ..writeAsBytesSync(bytes!.buffer.asUint8List());
    File('$out/$name.box').writeAsStringSync(
      '${topLeft.dx} ${topLeft.dy} ${render.size.width} ${render.size.height} $dpr\n',
    );
  });
}

void main() {
  for (final platform in [TargetPlatform.android, TargetPlatform.iOS]) {
    for (final dark in [false, true]) {
      for (final id in controls) {
        final tag = '${platform.name}-${dark ? 'dark' : 'light'}-$id';
        testWidgets(tag, (tester) async {
          debugDefaultTargetPlatformOverride = platform;
          tester.view.physicalSize = Size(400 * dpr, 400 * dpr);
          tester.view.devicePixelRatio = dpr;
          final route = '/c/$id${dark ? '/dark' : ''}';
          tester.binding.platformDispatcher.defaultRouteNameTestValue = route;
          await tester.pumpWidget(const ProbeApp(key: ValueKey(1)));
          await tester.pumpAndSettle();
          await save(tester, '$tag-u');
          final s0 = status(tester);
          for (var i = 0; i < 3 && !status(tester).contains('f=1'); i++) {
            await tester.sendKeyEvent(LogicalKeyboardKey.tab);
            await tester.pumpAndSettle(const Duration(milliseconds: 100));
          }
          await tester.pump(const Duration(milliseconds: 1200));
          await save(tester, '$tag-f1');
          final s1 = status(tester);
          await tester.pump(const Duration(milliseconds: 700));
          await save(tester, '$tag-f2');
          // A touch elsewhere switches the highlight mode back to touch.
          await tester.tapAt(const Offset(390, 390));
          await tester.pumpAndSettle();
          await tester.pump(const Duration(milliseconds: 1200));
          await save(tester, '$tag-e');
          final s2 = status(tester);
          // A fresh app, and a tap on the control.
          await tester.pumpWidget(const SizedBox());
          await tester.pumpWidget(const ProbeApp(key: ValueKey(2)));
          await tester.pumpAndSettle();
          final c = tester.getCenter(box);
          await tester.tapAt(c);
          await tester.pump(const Duration(milliseconds: 1500));
          await tester.pumpAndSettle();
          await save(tester, '$tag-t');
          final s3 = status(tester);
          // ignore: avoid_print
          print('V14W $tag | u: $s0 | tab: $s1 | touch elsewhere: $s2 | tap: $s3');
          File('$out/$tag.status').writeAsStringSync(
            'u: $s0\ntab: $s1\ntouch elsewhere: $s2\ntap: $s3\n',
          );
          tester.binding.platformDispatcher.clearDefaultRouteNameTestValue();
          tester.view.reset();
          debugDefaultTargetPlatformOverride = null;
        });
      }
    }
  }
}
