// Throwaway: V13 probe round 3 on the web. Not committed; kept as
// docs/evidence/probes/v13/v13_probe3.spec.js.txt.
//
// 1. A phone's browser, as far as Chromium can play one: the Pixel 7
//    descriptor (Android user agent, touch, 2.625 device pixel ratio), and
//    navigator.platform set to "Linux armv8l", which is what the engine
//    reads first (browser_detection.dart). Geometry and mouse clicks.
// 2. How many activations one touch tap at a fixture's centre gives, with
//    and without that platform: round 1 counted every touch near a fixture
//    as two.
import { test, devices } from '@playwright/test';

const byId = (page, x) => page.locator(`[flt-semantics-identifier="${x}"]`);
const { defaultBrowserType, ...pixel } = devices['Pixel 7'];

async function start(page) {
  await page.goto('about:blank');
  await page.goto('/');
  await byId(page, 'geo').waitFor();
  await page.waitForTimeout(1500);
}

async function geo(page) {
  const text = (await byId(page, 'geo').textContent()) ?? '';
  const parts = text.split(' | ');
  const fixtures = {};
  for (const part of parts.slice(1)) {
    const words = part.split(' ');
    const rects = {};
    for (const w of words.slice(1)) {
      const [k, v] = w.split('=');
      if (v === '-') continue;
      const [x, y, wd, ht] = v.split(',').map(Number);
      rects[k] = { x, y, r: x + wd, b: y + ht };
    }
    fixtures[words[0]] = rects;
  }
  return { header: parts[0], fixtures };
}

async function logCount(page) {
  const text = (await byId(page, 'log').textContent()) ?? '';
  const m = /Log\((\d+)\)/.exec(text);
  return m ? Number(m[1]) : -1;
}

const fmt = (r) => `${(r.r - r.x).toFixed(1)}x${(r.b - r.y).toFixed(1)}`;

async function centreTaps(page, label, tap) {
  const { fixtures } = await geo(page);
  const counts = [];
  for (const [id, rects] of Object.entries(fixtures)) {
    const s = rects.sem;
    const before = await logCount(page);
    await tap((s.x + s.r) / 2, (s.y + s.b) / 2);
    await page.waitForTimeout(900);
    counts.push(`${id}+${(await logCount(page)) - before}`);
  }
  console.log(`${label}: activations from one tap at each centre: ${counts.join(' ')}`);
}

test.describe('phone platform', () => {
  test.use(pixel);
  test.beforeEach(async ({ page }) => {
    await page.addInitScript(() => {
      Object.defineProperty(Navigator.prototype, 'platform', { get: () => 'Linux armv8l' });
    });
  });

  test('geometry and clicks', async ({ page }) => {
    test.setTimeout(300_000);
    await start(page);
    const { header, fixtures } = await geo(page);
    console.log(`== Pixel 7 with navigator.platform "${await page.evaluate(() => navigator.platform)}": ${header}`);
    for (const [id, rects] of Object.entries(fixtures)) {
      console.log(`${id} ${['lay', 'vis', 'shp', 'sem'].map((k) => `${k}=${rects[k] ? fmt(rects[k]) : '-'}`).join(' ')}`);
    }
    await centreTaps(page, 'phone platform, mouse click', (x, y) => page.mouse.click(x, y));
    await centreTaps(page, 'phone platform, touch tap', (x, y) => page.touchscreen.tap(x, y));
  });
});

test.describe('desktop platform with touch', () => {
  test.use(pixel);
  test('one touch at each centre', async ({ page }) => {
    test.setTimeout(300_000);
    await start(page);
    console.log(`== Pixel 7, navigator.platform "${await page.evaluate(() => navigator.platform)}": ${(await geo(page)).header}`);
    await centreTaps(page, 'desktop platform, touch tap', (x, y) => page.touchscreen.tap(x, y));
    await centreTaps(page, 'desktop platform, mouse click', (x, y) => page.mouse.click(x, y));
  });
});
