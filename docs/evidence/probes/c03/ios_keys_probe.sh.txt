#!/bin/sh
# C03 review probe on iOS, under tool/with_devices.py, from the worktree root,
# after `xcodebuild build-for-testing` into build/ios_uitest. Kept as
# docs/evidence/probes/c03/ios_keys_probe.sh.txt.
xcrun simctl list devices booted | grep -i "iphone"
cd ios || exit 1
xcodebuild test-without-building -workspace Runner.xcworkspace -scheme Runner \
  -destination "platform=iOS Simulator,name=iPhone 17" \
  -only-testing:"RunnerUITests/ProbeC03KeysUITests/testKeys" \
  -derivedDataPath ../build/ios_uitest 2>&1 | grep -E "C03KEYPROBE|error:|passed|failed" | grep -v "Test Suite"
xcrun simctl terminate booted dev.fluttera11y.flutterA11yTechniques 2>/dev/null
exit 0
