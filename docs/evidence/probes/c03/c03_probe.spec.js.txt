// C03 probe on the web (throwaway, not committed; kept as
// docs/evidence/probes/c03/c03_probe.spec.js.txt with c03probe.config.js.txt).
// For each fixture of lib/probe.dart: its element in the semantics DOM,
// Chromium's accessibility tree, and what a click, Enter, and Space do, each
// from a fresh load. A key that does nothing is followed, in the same load,
// by a click, the positive control. https://example.com is answered with a
// stub page, so that following it can be seen.
import { test } from '@playwright/test';

const ids = Array.from({ length: 14 }, (_, i) => `p${String(i + 1).padStart(2, '0')}`);

function log(...a) {
  console.log('C03WEB', ...a);
}

async function fresh(page, id) {
  await page.goto('about:blank');
  await page.goto(`/#/${id}`);
  await page.locator(`[flt-semantics-identifier="${id}-taps"]`).waitFor();
  await page.waitForTimeout(800);
}

// The deepest element whose aria-label or own text starts with the id,
// other than "Before pNN" and the taps line.
async function fixture(page, id) {
  const handle = await page.evaluateHandle((id) => {
    const all = [...document.querySelectorAll('flt-semantics-host *')];
    const match = (e) => {
      const name = e.getAttribute('aria-label') || [...e.childNodes].filter((n) => n.nodeType === 3).map((n) => n.textContent).join('');
      return name.startsWith(`${id} `) && !name.startsWith(`${id} taps`);
    };
    const found = all.filter(match);
    // The outermost element that holds the match and has a role, an href, a
    // tabindex, or is an anchor; else the match itself.
    let e = found[0];
    if (!e) return null;
    let up = e;
    while (up && up.tagName !== 'FLT-SEMANTICS-HOST') {
      if (up.tagName === 'A' || up.getAttribute('role') || up.hasAttribute('tabindex') || up.hasAttribute('flt-tappable')) {
        e = up;
        break;
      }
      up = up.parentElement;
    }
    return e;
  }, id);
  return handle.asElement();
}

async function state(page, context, id) {
  await page.waitForTimeout(1200);
  const url = page.url();
  let taps = '';
  let dest = false;
  if (!url.startsWith('https://example.com')) {
    taps = await page.locator(`[flt-semantics-identifier="${id}-taps"]`).textContent({ timeout: 1000 }).catch(() => '(gone)');
    dest = await page.getByText(/^Arrived/).count() > 0;
  }
  return `url=${url.replace(/^http:\/\/127\.0\.0\.1:\d+/, '')} taps="${taps}" destination=${dest} pages=${context.pages().length}`;
}

async function tabTo(page, el) {
  for (let i = 1; i <= 8; i++) {
    await page.keyboard.press('Tab');
    await page.waitForTimeout(150);
    const on = await el.evaluate((e) => e === document.activeElement || e.contains(document.activeElement));
    if (on) return i;
  }
  return 0;
}

test.describe.configure({ mode: 'serial' });

for (const id of ids) {
  test(`probe ${id}`, async ({ page, context }) => {
    await context.route('https://example.com/**', (r) => r.fulfill({ contentType: 'text/html', body: '<title>stub</title><p>stub page</p>' }));
    await fresh(page, id);
    const el = await fixture(page, id);
    if (!el) {
      log(id, 'NO ELEMENT FOUND');
      log(id, 'host', await page.locator('flt-semantics-host').evaluate((e) => e.outerHTML.replace(/ style="[^"]*"/g, '')));
      return;
    }
    const html = await el.evaluate((e) => e.outerHTML.replace(/ style="[^"]*"/g, ''));
    log(id, 'element', html);
    const cdp = await context.newCDPSession(page);
    const { nodes } = await cdp.send('Accessibility.getFullAXTree');
    const interesting = nodes.filter((n) => !n.ignored && (n.name?.value ?? '').startsWith(id));
    for (const n of interesting) {
      log(id, 'ax', `role=${n.role?.value} name="${n.name?.value}" props=${(n.properties ?? []).map((p) => `${p.name}:${JSON.stringify(p.value?.value)}`).join(',')}`);
    }
    log(id, 'aria', JSON.stringify(await page.locator('flt-semantics-host').ariaSnapshot()));

    // Click with the mouse, at the element's centre.
    const box = await el.boundingBox();
    await page.mouse.click(box.x + box.width / 2, box.y + box.height / 2);
    log(id, 'click', await state(page, context, id));
    for (const p of context.pages().slice(1)) await p.close();

    for (const key of ['Enter', 'Space']) {
      await fresh(page, id);
      const target = await fixture(page, id);
      const reached = await tabTo(page, target);
      log(id, `tab reaches it after ${reached} Tab(s)`);
      if (!reached) continue;
      await page.keyboard.press(key);
      const after = await state(page, context, id);
      log(id, key, after);
      for (const p of context.pages().slice(1)) await p.close();
      if (!page.url().startsWith('https://example.com') && !/destination=true/.test(after)) {
        const again = await fixture(page, id);
        const b = await again.boundingBox();
        await page.mouse.click(b.x + b.width / 2, b.y + b.height / 2);
        log(id, `${key} control click`, await state(page, context, id));
        for (const p of context.pages().slice(1)) await p.close();
      }
    }
    // Command-click (Meta on macOS): a new tab, as for a link?
    await fresh(page, id);
    const t2 = await fixture(page, id);
    const b2 = await t2.boundingBox();
    await page.keyboard.down('Meta');
    await page.mouse.click(b2.x + b2.width / 2, b2.y + b2.height / 2);
    await page.keyboard.up('Meta');
    log(id, 'meta-click', await state(page, context, id), 'other pages:', context.pages().slice(1).map((p) => p.url()).join(' '));
    for (const p of context.pages().slice(1)) await p.close();
  });
}
