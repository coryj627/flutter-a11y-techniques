// C03 probe (throwaway, not committed; kept as
// docs/evidence/probes/c03/probe.dart.txt): what a stand-alone link built
// from the SDK exposes, and what activates it. One fixture per route,
// `/p01` to `/p14`; `/` lists them all; `/dest` is an in-app destination
// that the fixtures with `linkUrl: '#/dest'` point to.
//
// Every fixture's label starts with its id. A tap that reaches a fixture's
// handler prints "C03PROBE TAP pNN" and adds one to the "pNN taps: N" text
// under it. Opening `/dest` prints "C03PROBE DEST" and the number of times
// it has been opened.
import 'package:flutter/cupertino.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';
import 'package:flutter/services.dart';
import 'package:url_launcher/link.dart' as ul;

import 'techniques/s02_informative_images/example_image.dart';

void main() {
  runApp(const ProbeApp());
  SemanticsBinding.instance.ensureSemantics();
}

final ext = Uri.parse('https://example.com/c03');
final dest = Uri.parse('#/dest');
int destOpened = 0;

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      onGenerateRoute: (settings) {
        final name = settings.name ?? '/';
        Widget page;
        if (name == '/dest') {
          destOpened++;
          debugPrint('C03PROBE DEST $destOpened');
          page = Scaffold(
            appBar: AppBar(title: const Text('Destination')),
            body: Text('Arrived $destOpened'),
          );
        } else if (fixtures.containsKey(name.replaceFirst('/', ''))) {
          final id = name.replaceFirst('/', '');
          page = FixtureScreen(ids: [id]);
        } else {
          page = FixtureScreen(ids: fixtures.keys.toList());
        }
        return MaterialPageRoute<void>(settings: settings, builder: (_) => page);
      },
    );
  }
}

typedef Build = Widget Function(BuildContext context, VoidCallback tap);

Widget _img({String? label, bool exclude = false}) => Image.memory(
  sunrisePng,
  width: 72,
  height: 48,
  fit: BoxFit.cover,
  semanticLabel: label,
  excludeFromSemantics: exclude,
);

TextStyle _linkStyle(BuildContext context) {
  final c = Theme.of(context).colorScheme.primary;
  return TextStyle(
    color: c,
    decoration: TextDecoration.underline,
    decorationColor: c,
  );
}

class FixtureScreen extends StatefulWidget {
  const FixtureScreen({super.key, required this.ids});

  final List<String> ids;

  @override
  State<FixtureScreen> createState() => _FixtureScreenState();
}

class _FixtureScreenState extends State<FixtureScreen> {
  final taps = <String, int>{};

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.ids.length == 1 ? widget.ids.single : 'C03 probe')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          spacing: 12,
          children: [
            for (final id in widget.ids) ...[
              Text('Before $id'),
              Builder(
                builder: (context) => fixtures[id]!.$2(context, () {
                  debugPrint('C03PROBE TAP $id');
                  setState(() => taps[id] = (taps[id] ?? 0) + 1);
                }),
              ),
              Semantics(
                identifier: '$id-taps',
                child: Text('$id taps: ${taps[id] ?? 0}'),
              ),
            ],
          ],
        ),
      ),
    );
  }
}
