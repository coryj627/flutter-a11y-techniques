// C12 probe, fourth part (throwaway; kept as
// docs/evidence/probes/c12/probe4_test.dart.txt): the text contrast
// guideline on Material buttons with keyboard focus, a hover, and both, in
// light and dark, after good-description's FilledButton came out under
// 4.5:1 with focus and a hover.
import 'dart:ui';

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';

void main() {
  final buttons = <String, Widget Function()>{
    'FilledButton': () => FilledButton(onPressed: () {}, child: const Text('Download')),
    'OutlinedButton': () => OutlinedButton(onPressed: () {}, child: const Text('Download')),
    'TextButton': () => TextButton(onPressed: () {}, child: const Text('Download')),
  };
  for (final MapEntry(key: name, value: build) in buttons.entries) {
    for (final b in Brightness.values) {
      for (final (focus, hover) in [(false, false), (true, false), (false, true), (true, true)]) {
        testWidgets('$name ${b.name} focus=$focus hover=$hover', (tester) async {
          final h = tester.ensureSemantics();
          await tester.pumpWidget(MaterialApp(
            theme: ThemeData(colorSchemeSeed: Colors.indigo, brightness: b),
            home: Scaffold(body: Center(child: build())),
          ));
          if (focus) {
            await tester.sendKeyEvent(LogicalKeyboardKey.tab);
          }
          if (hover) {
            final m = await tester.createGesture(kind: PointerDeviceKind.mouse);
            await m.addPointer(location: tester.getCenter(find.text('Download')));
            addTearDown(m.removePointer);
          }
          await tester.pumpAndSettle();
          final result = await textContrastGuideline.evaluate(tester);
          debugPrint('C12TEST $name ${b.name} focus=$focus hover=$hover: '
              '${result.passed ? 'passes' : result.reason?.split('\n').firstWhere((l) => l.contains('contrast ratio'), orElse: () => 'fails')}');
          h.dispose();
        });
      }
    }
  }
}
