// S12 probe on the web (throwaway, not committed; kept as
// docs/evidence/probes/s12/s12_probe.spec.js.txt with s12probe.config.js.txt).
// For each fixture of lib/probe.dart: the semantics DOM between its
// "Before" text and its taps line, Chromium's accessibility tree over the
// same span, every stop of six Tabs (the DOM's active element, and the
// taps line after); for the fixtures with a Share button, Enter after one
// to three Tabs (and Space for some), each from a fresh load, and a mouse
// click where the Share button is laid out. Then the plain HTML comparison
// page, html.html (html.html.txt): Chromium's tree and the Tab stops.
import { test } from '@playwright/test';

const ids = Array.from({ length: 34 }, (_, i) => `p${String(i + 1).padStart(2, '0')}`);
const withShare = [
  'p01', 'p03', 'p04', 'p05', 'p07', 'p09', 'p10', 'p11', 'p12', 'p13', 'p14', 'p15', 'p16',
  'p17', 'p18', 'p19', 'p20', 'p22', 'p23', 'p24', 'p29', 'p31', 'p32',
];
const withSpace = ['p03', 'p05', 'p13', 'p16', 'p22', 'p31'];

function log(...a) {
  console.log('S12WEB', ...a.map((x) => String(x).replace(/\n/g, '\\n')));
}

async function fresh(page, id) {
  await page.goto('about:blank');
  await page.goto(`/#/${id}`);
  await page.locator(`[flt-semantics-identifier="${id}-taps"]`).waitFor();
  await page.waitForTimeout(800);
}

function clean(html) {
  return html.replace(/ style="[^"]*"/g, '').replace(/ id="flt-semantic-node-\d+"/g, '');
}

async function taps(page, id) {
  await page.waitForTimeout(800);
  return page
    .locator(`[flt-semantics-identifier="${id}-taps"]`)
    .textContent({ timeout: 1000 })
    .catch(() => '(gone)');
}

async function active(page) {
  return page.evaluate(() => {
    const e = document.activeElement;
    if (!e) return '(none)';
    if (!e.closest('flt-semantics-host')) return `(page: ${e.tagName}${e.id ? '#' + e.id : ''})`;
    const role = e.getAttribute('role');
    return `${e.tagName}${role ? `[role=${role}]` : ''} "${(e.getAttribute('aria-label') || e.textContent.trim()).slice(0, 60)}"`;
  });
}

async function axTree(page, context, from, to) {
  const cdp = await context.newCDPSession(page);
  const { nodes } = await cdp.send('Accessibility.getFullAXTree');
  const byId = new Map(nodes.map((n) => [n.nodeId, n]));
  const depth = (n) => {
    let d = 0;
    let p = n.parentId;
    while (p && byId.get(p)) {
      d++;
      p = byId.get(p).parentId;
    }
    return d;
  };
  const out = [];
  let on = from === null;
  const root = nodes.find((n) => !n.parentId);
  const visit = (n) => {
    if (!n) return;
    const name = n.name?.value ?? '';
    if (!n.ignored && n.role?.value !== 'InlineTextBox') {
      if (from !== null && name === from) on = true;
      else if (to !== null && name.startsWith(to)) on = false;
      else if (on) {
        const props = (n.properties ?? []).map((p) => `${p.name}:${JSON.stringify(p.value?.value)}`).join(',');
        out.push(`${'  '.repeat(depth(n))}role=${n.role?.value} name="${name}" props=${props}`);
      }
    }
    for (const c of n.childIds ?? []) visit(byId.get(c));
  };
  visit(root);
  return out;
}

test.describe.configure({ mode: 'serial' });

for (const id of ids) {
  test(`probe ${id}`, async ({ page, context }) => {
    await fresh(page, id);
    const html = await page.evaluate((id) => {
      const host = document.querySelector('flt-semantics-host');
      const before = [...host.querySelectorAll('*')].find(
        (e) => e.textContent.trim() === `Before ${id}` && e.children.length === 0,
      );
      const after = host.querySelector(`[flt-semantics-identifier="${id}-taps"]`);
      if (!before) return ['(no Before element)', host.outerHTML];
      let a = before;
      while (a.parentElement && !a.parentElement.contains(after)) a = a.parentElement;
      const out = [];
      for (let s = a.nextElementSibling; s && !s.contains(after); s = s.nextElementSibling) out.push(s.outerHTML);
      return out;
    }, id);
    for (const h of html) log(id, 'element', clean(h));
    for (const l of await axTree(page, context, `Before ${id}`, `${id} taps`)) log(id, 'ax', l);
    const stops = [];
    for (let i = 0; i < 6; i++) {
      await page.keyboard.press('Tab');
      await page.waitForTimeout(200);
      stops.push(await active(page));
    }
    log(id, 'tab stops', stops.join(' -> '));
    if (!withShare.includes(id)) return;
    for (const key of withSpace.includes(id) ? ['Enter', 'Space'] : ['Enter']) {
      for (let n = 1; n <= 3; n++) {
        await fresh(page, id);
        for (let i = 0; i < n; i++) {
          await page.keyboard.press('Tab');
          await page.waitForTimeout(200);
        }
        const on = await active(page);
        await page.keyboard.press(key);
        log(id, `${n} Tab(s) (on ${on}) then ${key}:`, await taps(page, id));
        if ((await taps(page, id)) === '(gone)') log(id, 'the screen closed');
      }
    }
    await fresh(page, id);
    const at = await page.evaluate((id) => {
      const host = document.querySelector('flt-semantics-host');
      const find = (t) => [...host.querySelectorAll('*')].find(
        (e) => (e.getAttribute('aria-label') || e.textContent).trim() === t,
      );
      const before = find(`Before ${id}`);
      const after = find(`${id} After`);
      if (!before || !after) return null;
      const b = before.getBoundingClientRect();
      const a = after.getBoundingClientRect();
      return { x: a.left + 40, y: (b.bottom + a.top) / 2 };
    }, id);
    if (at) {
      await page.mouse.click(at.x, at.y);
      log(id, `click at ${Math.round(at.x)},${Math.round(at.y)} (between Before and After):`, await taps(page, id));
    }
  });
}

test('plain HTML comparison', async ({ page, context }) => {
  await page.goto('/html.html');
  await page.waitForTimeout(500);
  for (const l of await axTree(page, context, null, null)) log('html', 'ax', l);
  const stops = [];
  for (let i = 0; i < 10; i++) {
    await page.keyboard.press('Tab');
    await page.waitForTimeout(100);
    stops.push(await page.evaluate(() => document.activeElement?.id || document.activeElement?.tagName));
  }
  log('html', 'tab stops', stops.join(' -> '));
  for (const id of ['b-aria-hidden', 'b-inert', 'b-hidden', 'b-opacity', 'b-visibility']) {
    const r = await page.evaluate((id) => {
      const e = document.getElementById(id);
      e.focus();
      return document.activeElement === e ? 'focused' : `not focused (active: ${document.activeElement?.id || document.activeElement?.tagName})`;
    }, id);
    log('html', `focus() on #${id}:`, r);
  }
});
