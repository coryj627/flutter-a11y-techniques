// S12 probe (throwaway, not committed; kept as
// docs/evidence/probes/s12/probe.dart.txt): what each of Flutter's ways to
// hide content leaves to assistive technology and to the keyboard. One
// fixture per route, `/p01` to `/p34`; `/` lists nothing.
//
// Each fixture sits on a screen laid out as the app's own technique screens
// are (a `Scaffold` whose body is a `SingleChildScrollView` round a
// `Column`), between a "Before pNN" text and a "pNN taps: a=N b=N" text,
// which has the test identifier `pNN-taps`. `a` counts the taps on the
// fixture's own button ("pNN Share"), where it has one; `b` counts the taps
// on the "pNN After" button, which follows every fixture and is never
// hidden: the positive control for the keyboard and the screen reader.
// Every text starts with the fixture's id.
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';

void main() {
  runApp(const ProbeApp());
  SemanticsBinding.instance.ensureSemantics();
}

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      onGenerateRoute: (settings) {
        final name = (settings.name ?? '/').replaceFirst('/', '');
        final page = fixtures.containsKey(name)
            ? FixtureScreen(id: name)
            : const Scaffold(body: Text('S12 probe'));
        return MaterialPageRoute<void>(
          settings: settings,
          builder: (_) => page,
        );
      },
    );
  }
}

typedef Build = Widget Function(String id, void Function() tap);

class FixtureScreen extends StatefulWidget {
  const FixtureScreen({super.key, required this.id});

  final String id;

  @override
  State<FixtureScreen> createState() => _FixtureScreenState();
}

class _FixtureScreenState extends State<FixtureScreen> {
  var a = 0;
  var b = 0;

  @override
  Widget build(BuildContext context) {
    final id = widget.id;
    return Scaffold(
      appBar: AppBar(title: Text('Probe $id')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          spacing: 12,
          children: [
            Text('Before $id'),
            fixtures[id]!.$2(id, () => setState(() => a++)),
            TextButton(
              onPressed: () => setState(() => b++),
              child: Text('$id After'),
            ),
            Semantics(
              identifier: '$id-taps',
              child: Text('$id taps: a=$a b=$b'),
            ),
          ],
        ),
      ),
    );
  }
}
