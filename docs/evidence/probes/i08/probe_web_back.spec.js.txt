// Throwaway: I08 round two probe of the browser's Back on the web
// (Chromium, Playwright). Not committed; kept as
// docs/evidence/probes/i08/probe_web_back.spec.js.txt. For each example and
// each way of going back, from each switch: show the layers from the
// keyboard, Tab to the switch, go back, and record whether both switches
// are gone, the example's count, where DOM focus is, the page's URL, and
// whether the technique's screen is shown instead.
import { test } from '@playwright/test';

const byId = (page, id) => page.locator(`[flt-semantics-identifier="${id}"]`);

const stop = (page, slug) => page.evaluate((prefix) => {
  const e = document.activeElement;
  if (!e || e === document.body || e.tagName === 'FLUTTER-VIEW') return 'outside';
  const id = e.closest('[flt-semantics-identifier]')?.getAttribute('flt-semantics-identifier');
  if (id && id.startsWith(prefix)) return id.slice(prefix.length);
  return `other: ${e.tagName} ${e.getAttribute('role') ?? ''} ${e.textContent.trim().slice(0, 30)}`;
}, `i08-${slug}-`);

async function press(page, key) {
  await page.keyboard.press(key);
  await page.waitForTimeout(400);
}

async function state(page, slug) {
  const switches = (await byId(page, `i08-${slug}-contours`).count()) + (await byId(page, `i08-${slug}-campsites`).count());
  const count = (await byId(page, `i08-${slug}-count`).count()) ? (await byId(page, `i08-${slug}-count`).textContent()).trim() : '(no count)';
  const technique = await page.getByRole('heading', { name: 'Dismiss and escape', level: 1 }).count();
  return `switches ${switches}, ${count}, focus ${await stop(page, slug)}, technique screen ${technique ? 'shown' : 'not shown'}, url ${page.url().replace(/^https?:\/\/[^/]+/, '')}`;
}

const routes = ['good-material', 'bad-not-dismissible'];
const ways = {
  'Alt+ArrowLeft': (page) => press(page, 'Alt+ArrowLeft'),
  'Meta+BracketLeft': (page) => press(page, 'Meta+BracketLeft'),
  'page.goBack()': async (page) => { await page.goBack({ waitUntil: 'commit' }).catch(() => {}); await page.waitForTimeout(800); },
};

for (const slug of ['bad-no-escape', 'bad-not-dismissible', 'good-widgets', 'good-material', 'bad-back-leaves']) {
  for (const [way, back] of Object.entries(ways)) {
    for (const part of ['contours', 'campsites']) {
      test(`${slug} ${way} from ${part}`, async ({ page }) => {
        await page.goto('about:blank');
        await page.goto(`/#/t/I08/${slug}`);
        await byId(page, `i08-${slug}-count`).waitFor();
        await page.waitForTimeout(600);
        await press(page, 'Tab');
        await press(page, 'Tab');
        await press(page, 'Enter');
        if (routes.includes(slug)) await press(page, 'Tab');
        if (part === 'campsites') await press(page, 'Tab');
        const before = await state(page, slug);
        const historyLength = await page.evaluate(() => history.length);
        await back(page);
        await page.waitForTimeout(800);
        const after = await state(page, slug);
        console.log(`BACK ${slug} | ${way} | from ${part} | before: ${before} | history.length ${historyLength} | after: ${after}`);
      });
    }
  }
}
