#!/bin/sh
# I08 probe on iOS, under tool/with_devices.py. Run from the worktree root,
# after `fvm flutter build ios --simulator --debug -t lib/probe.dart` and
# `xcodebuild build-for-testing` into build/ios_uitest. VoiceOver is turned
# on only inside testVoiceOverEscapeKey, through XCUIVoiceOverService, and
# off again at the end of each fixture.
cd ios || exit 1
echo "== simulator: $(xcrun simctl list devices booted | grep -E 'Booted' | head -1)"
run() {
  echo "== $1"
  xcodebuild test-without-building -workspace Runner.xcworkspace -scheme Runner \
    -destination "platform=iOS Simulator,name=iPhone 17" \
    -only-testing:"RunnerUITests/I08ProbeUITests/$1" \
    -derivedDataPath ../build/ios_uitest 2>&1 | grep -E "I08PROBE|error:|passed|failed|skipped" | grep -v "Test Suite"
}
for m in ${METHODS:-testBackControl testSurfaces testTooltip testTraps testVoiceOverEscapeKey}; do run $m; done
xcrun simctl terminate booted dev.fluttera11y.flutterA11yTechniques 2>/dev/null
exit 0
