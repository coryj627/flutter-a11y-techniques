// Throwaway: I08 step-4 probe. Can a tooltip kept open by hover, while the
// keyboard opens the example and its layers, make Escape fail without CPU
// throttling? Not committed; kept as
// docs/evidence/probes/i08/probe_web_flake_hover.spec.js.txt.
//
// On the technique screen the mouse rests on Back until its tooltip is
// shown; then, with the mouse left there, Tab goes to good-widgets' title
// and Enter opens it; Tab, Tab, Enter shows the layers; once DOM focus is
// on Contour lines, Escape, once. Records whether the layers closed within
// five seconds, and whether a tooltip was shown at each step (an element
// with the text "Back" near the app bar that is not the Back button).
// P08_PLATFORM=linux as in probe_web_flake.spec.js.txt.
import fs from 'node:fs';
import { test } from '@playwright/test';

const out = process.env.P08_OUT ?? 'p08_det.jsonl';
const platform = process.env.P08_PLATFORM ?? 'native';
const variant = process.env.P08_VARIANT ?? 'hover';
// P08_GOOD=good-material runs the same on the Material sheet (a route: Tab
// from the route's scope to Contour lines).
const good = process.env.P08_GOOD ?? 'good-widgets';
const titles = {
  'good-widgets': /^Layers drawn in the overlay, closed by every dismiss command/,
  'good-material': /^A Material bottom sheet that can be dismissed/,
  'good-cupertino': /^A Cupertino modal popup with a dismissible barrier/,
};
const byId = (page, id) => page.locator(`[flt-semantics-identifier="${id}"]`);

const tooltip = (page) => page.evaluate(() => [...document.querySelectorAll('flt-semantics-host *')]
  .filter((e) => [...e.childNodes].some((n) => n.nodeType === 3 && n.textContent.trim() === 'Back'))
  .filter((e) => e.closest('flt-semantics')?.getAttribute('role') !== 'button')
  .filter((e) => { const r = e.getBoundingClientRect(); return r.y >= 0 && r.y < 120; }).length > 0);

const stop = (page) => page.evaluate(() => {
  const e = document.activeElement;
  return e?.closest('[flt-semantics-identifier]')?.getAttribute('flt-semantics-identifier') ?? `${e?.tagName} ${e?.textContent?.trim().slice(0, 20)}`;
});

for (let run = 0; run < Number(process.env.P08_RUNS ?? 1); run++) {
  test(`hover then keyboard, run ${run}`, async ({ page }) => {
    if (platform === 'linux') {
      await page.addInitScript(() => Object.defineProperty(Navigator.prototype, 'platform', { get: () => 'Linux x86_64' }));
    }
    const steps = [];
    await page.goto('about:blank');
    await page.goto('/#/t/I08');
    await page.getByRole('heading', { name: 'Dismiss and escape', level: 1 }).waitFor();
    await page.waitForTimeout(500);
    const back = await page.getByRole('button', { name: 'Back', exact: true }).boundingBox();
    await page.mouse.move(back.x + back.width / 2, back.y + back.height / 2);
    await page.waitForTimeout(1000);
    steps.push(`hovered Back: tooltip ${await tooltip(page)}`);
    const title = page.getByRole('button', { name: titles[good] });
    let found = false;
    for (let i = 0; i < 12 && !found; i++) {
      await page.keyboard.press('Tab');
      await page.waitForTimeout(150);
      found = await title.evaluate((e) => e === document.activeElement || e.contains(document.activeElement));
    }
    steps.push(`title focused ${found}; tooltip ${await tooltip(page)}`);
    await page.keyboard.press('Enter');
    await byId(page, `i08-${good}-count`).waitFor();
    await page.waitForTimeout(variant === 'quick' ? 0 : 800);
    steps.push(`example shown: tooltip ${await tooltip(page)}; focus ${await stop(page)}`);
    for (let i = 0; i < 6 && (await stop(page)) !== `i08-${good}-open`; i++) {
      await page.keyboard.press('Tab');
      await page.waitForTimeout(200);
    }
    steps.push(`before Enter: focus ${await stop(page)}`);
    await page.keyboard.press('Enter');
    await byId(page, `i08-${good}-contours`).waitFor();
    const start = Date.now();
    while ((await stop(page)) !== `i08-${good}-contours` && Date.now() - start < 5000) await page.waitForTimeout(50);
    if (good !== 'good-widgets') {
      await page.waitForTimeout(600);
      await page.keyboard.press('Tab');
      await page.waitForTimeout(400);
    }
    steps.push(`layers shown: focus ${await stop(page)}`);
    const delay = Number(process.env.P08_DELAY ?? 0);
    if (delay) await page.waitForTimeout(delay);
    steps.push(`waited ${delay} ms before Escape`);
    const dump = () => page.evaluate(() => (window.p08dump ? window.p08dump() : null));
    steps.push(`dump before Escape: ${await dump()}`);
    await page.keyboard.press('Escape');
    const at = Date.now();
    let closed = false;
    while (Date.now() - at < 5000) {
      if ((await byId(page, `i08-${good}-contours`).count()) === 0) { closed = true; break; }
      await page.waitForTimeout(50);
    }
    steps.push(`dump after Escape: ${await dump()}`);
    // Diagnostic only, after the outcome: a second Escape a second later.
    let second = null;
    if (!closed) {
      await page.waitForTimeout(1000);
      await page.keyboard.press('Escape');
      await page.waitForTimeout(2000);
      second = (await byId(page, `i08-${good}-contours`).count()) === 0 ? 'closed' : 'still shown';
    }
    fs.appendFileSync(out, JSON.stringify({ platform, variant, good, delay: Number(process.env.P08_DELAY ?? 0), run, closed, second, steps }) + '\n');
  });
}
