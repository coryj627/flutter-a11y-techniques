// Throwaway: I08 probe on the web (Chromium, Playwright). Not committed;
// kept as docs/evidence/probes/i08/probe_web.spec.js.txt. For each fixture
// of lib/probe.dart: open it with a click, or with DOM focus on its button
// and Enter, then press Escape, and read the probe's status bar ("Open:",
// "KF:", "Log:") and document.activeElement. Then the traps: Tab and
// Shift+Tab from inside each, then Escape.
import { test } from '@playwright/test';

const id = (page, x) => page.locator(`[flt-semantics-identifier="${x}"]`);
const status = async (page, x) =>
  (await id(page, x).getAttribute('aria-label')) ?? (await id(page, x).textContent()) ?? '';
const active = (page) => page.evaluate(() => {
  const e = document.activeElement;
  if (!e || e === document.body) return 'body';
  if (e.tagName === 'FLUTTER-VIEW') return 'flutter-view';
  const ident = e.getAttribute('flt-semantics-identifier');
  if (ident) return ident;
  const role = e.getAttribute('role') ?? '';
  return `${e.tagName.toLowerCase()}{role=${role},label=${e.getAttribute('aria-label') ?? ''},text=${(e.textContent ?? '').trim().slice(0, 24)}}`;
});
const bar = async (page) =>
  `${await status(page, 'open')} / ${await status(page, 'kf')} / DOM ${await active(page)} / ${await status(page, 'log')}`;

async function start(page, route = '/probe', ready = 'e0 traps screen') {
  await page.goto('about:blank');
  await page.goto(`/#${route}`);
  await id(page, ready).waitFor();
  await page.waitForTimeout(800);
}

async function press(page, key) {
  await page.keyboard.press(key);
  await page.waitForTimeout(500);
}

const fixtures = {
  e1: 'e1 showDialog',
  e2: 'e2 showDialog not dismissible',
  e3: 'e3 showCupertinoDialog',
  e4: 'e4 showCupertinoDialog dismissible',
  e5: 'e5 showModalBottomSheet',
  e6: 'e6 showModalBottomSheet not dismissible',
  e7: 'e7 persistent sheet',
  e8: 'e8 MenuAnchor',
  e9: null,
  e11: 'e11 open drawer',
  e12: 'e12 SnackBar',
  e13: 'e13 showDatePicker',
  e14: 'e14 OverlayEntry',
  e15: 'e15 OverlayPortal',
  e16: 'e16 showGeneralDialog',
  e17: 'e17 showGeneralDialog not dismissible',
  e18: 'e18 showCupertinoModalPopup',
  e19: 'e19 showDialog PopScope',
  e20: 'e20 OverlayPortal handled',
  e21: null,
  e22: 'e22 showCupertinoSheet',
};

const trigger = (page, key) => {
  if (key === 'e9') return page.getByRole('button', { name: 'e9 PopupMenuButton' }).first();
  if (key === 'e21') return page.getByRole('button', { name: /e21 DropdownButton/ }).first();
  return page.getByRole('button', { name: fixtures[key], exact: true });
};

for (const key of Object.keys(fixtures)) {
  for (const how of ['click', 'keys']) {
    test(`${key} ${how}`, async ({ page }) => {
      await start(page);
      const t = trigger(page, key);
      await t.scrollIntoViewIfNeeded();
      if (how === 'click') {
        await t.click();
      } else {
        await t.focus();
        await page.waitForTimeout(300);
        await press(page, 'Enter');
      }
      await page.waitForTimeout(1200);
      const modal = await page.locator('[aria-modal="true"]').count();
      const dialogs = await page.locator('[role="dialog"], [role="alertdialog"]').count();
      const opened = `${await bar(page)}${key === 'e21' ? ` / e21 items ${await page.getByRole('menuitem', { name: 'e21 One' }).count()}` : ''}`;
      await press(page, 'Escape');
      await page.waitForTimeout(800);
      const after = `${await bar(page)}${key === 'e21' ? ` / e21 items ${await page.getByRole('menuitem', { name: 'e21 One' }).count()}` : ''}`;
      console.log(`WEB ${key} ${how} | opened: ${opened} | aria-modal ${modal}, dialog roles ${dialogs} | Escape: ${after}`);
    });
  }
}

// The tooltip: shown by hovering, then Escape, with DOM focus on the page's
// body and with it on the tooltip's own button.
for (const focus of ['none', 'button']) {
  test(`e10 hover, focus on ${focus}`, async ({ page }) => {
    await start(page);
    const button = page.getByRole('button', { name: 'e10 Tooltip', exact: true });
    await button.scrollIntoViewIfNeeded();
    if (focus === 'button') {
      await button.focus();
      await page.waitForTimeout(300);
    } else {
      await page.evaluate(() => document.activeElement?.blur());
      await page.waitForTimeout(300);
    }
    await button.hover();
    await page.waitForTimeout(2000);
    const shown = await page.getByText('e10 tooltip text').count();
    const domBefore = await active(page);
    await press(page, 'Escape');
    await page.waitForTimeout(500);
    const after = await page.getByText('e10 tooltip text').count();
    // The pointer stays over the button: does it show again?
    await page.waitForTimeout(2000);
    const later = await page.getByText('e10 tooltip text').count();
    console.log(`WEB e10 hover focus=${focus} | DOM ${domBefore} | shown ${shown} | after Escape ${after} | 2 s later ${later} | ${await status(page, 'keys')}`);
  });
}

test('traps', async ({ page }) => {
  await start(page, '/traps', 't9 after');
  const out = [];
  await press(page, 'F6');
  const fromT0 = [await status(page, 'kf')];
  for (let i = 0; i < 4; i++) {
    await press(page, 'Tab');
    fromT0.push(`${await status(page, 'kf')} [${await active(page)}]`);
  }
  out.push(`from t0: ${fromT0.join(' > ')}`);
  for (const [t, f] of [['t1', 'F1'], ['t2', 'F2'], ['t3', 'F3'], ['t4', 'F4'], ['t5', 'F5']]) {
    await press(page, f);
    const fwd = [`${await status(page, 'kf')} [${await active(page)}]`];
    for (let i = 0; i < 4; i++) {
      await press(page, 'Tab');
      fwd.push(`${await status(page, 'kf')} [${await active(page)}]`);
    }
    const back = [];
    for (let i = 0; i < 4; i++) {
      await press(page, 'Shift+Tab');
      back.push(`${await status(page, 'kf')} [${await active(page)}]`);
    }
    await press(page, 'Escape');
    const esc = `${await status(page, 'kf')} [${await active(page)}]`;
    out.push(`${t} | Tab: ${fwd.join(' > ')} | Shift+Tab: ${back.join(' > ')} | Escape: ${esc}`);
  }
  console.log(`WEB traps\n${out.join('\n')}\n${await status(page, 'keys')}`);
});
