// I08 probe (throwaway, not committed; kept as
// docs/evidence/probes/i08/probe.dart.txt): what Escape, Back, and the
// screen readers' dismiss commands do to each temporary surface the SDK
// offers, and what Tab and Shift+Tab do in a focus trap made three ways.
//
// Every control's text starts with its fixture id. A status bar at the
// bottom of the screen, outside the Navigator so that no barrier hides it
// from the semantics tree, shows:
//   "Open: e1"            the fixtures open now;
//   "KF: node:e1 Close"   the node with Flutter's keyboard focus, named by
//                         the first text under it ("scope:..." for a
//                         FocusScopeNode with no focused child);
//   "Log: ..."            what happened: "e1 closed", "popRoute",
//                         "didPop e1" (a route the Navigator popped);
//   "Keys: ..."           the last keys that reached the framework.
import 'dart:async';

import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';
import 'package:flutter/services.dart';

void main() {
  WidgetsFlutterBinding.ensureInitialized();
  // Registered before the app's own observer, so it sees every popRoute
  // first; it handles none.
  WidgetsBinding.instance.addObserver(_PopLogger());
  // Above the app, so it sees every key that nothing in the app handled,
  // the app's own shortcuts included.
  runApp(
    Focus(
      canRequestFocus: false,
      skipTraversal: true,
      onKeyEvent: (node, event) {
        if (event is KeyDownEvent)
          log('unhandled ${event.logicalKey.keyLabel}');
        return KeyEventResult.ignored;
      },
      child: const ProbeApp(),
    ),
  );
  SemanticsBinding.instance.ensureSemantics();
}

final ValueNotifier<List<String>> probeLog = ValueNotifier<List<String>>([]);
final ValueNotifier<Set<String>> openNow = ValueNotifier<Set<String>>({});

void log(String text) {
  final next = [...probeLog.value, text];
  probeLog.value = next.length > 8 ? next.sublist(next.length - 8) : next;
}

void opened(String id) => openNow.value = {...openNow.value, id};

void closed(String id, [String how = '']) {
  openNow.value = {...openNow.value}..remove(id);
  log('$id closed$how');
}

class _PopLogger with WidgetsBindingObserver {
  @override
  Future<bool> didPopRoute() async {
    log('popRoute');
    return false;
  }
}

class _NavLog extends NavigatorObserver {
  @override
  void didPop(Route<dynamic> route, Route<dynamic>? previousRoute) {
    log('didPop ${route.settings.name ?? route.runtimeType}');
  }
}

/// The probe's own keys (see _StatusBarState._key): filled in by the trap
/// screen while it is up.
final Map<LogicalKeyboardKey, FocusNode> trapKeys = {};

String? firstText(Element element) {
  String? found;
  void visit(Element e) {
    if (found != null) return;
    final w = e.widget;
    if (w is Text && w.data != null) {
      found = w.data;
    } else if (w is Tooltip && w.message != null) {
      found = w.message;
    } else if (w is Semantics && w.properties.label != null) {
      found = w.properties.label;
    } else {
      e.visitChildren(visit);
    }
  }

  visit(element);
  return found;
}

String describeFocus(FocusNode? node) {
  if (node == null) return 'none';
  final kind = node is FocusScopeNode ? 'scope' : 'node';
  final ctx = node.context;
  final text = ctx is Element ? firstText(ctx) : null;
  return '$kind:${text ?? '?'}';
}

class StatusBar extends StatefulWidget {
  const StatusBar({super.key});

  @override
  State<StatusBar> createState() => _StatusBarState();
}

class _StatusBarState extends State<StatusBar> {
  String _kf = 'none';
  final List<String> _keys = [];

  @override
  void initState() {
    super.initState();
    FocusManager.instance.addListener(_focus);
    HardwareKeyboard.instance.addHandler(_key);
    probeLog.addListener(_changed);
    openNow.addListener(_changed);
  }

  @override
  void dispose() {
    FocusManager.instance.removeListener(_focus);
    HardwareKeyboard.instance.removeHandler(_key);
    probeLog.removeListener(_changed);
    openNow.removeListener(_changed);
    super.dispose();
  }

  void _changed() {
    if (mounted) setState(() {});
  }

  void _focus() {
    final next = describeFocus(FocusManager.instance.primaryFocus);
    if (next != _kf && mounted) setState(() => _kf = next);
  }

  bool _key(KeyEvent event) {
    // The probe's own keys, for the device runs: F1 to F5 put focus on the
    // first button of trap t1 to t5, F6 on "t0 before". They are taken
    // here, before any focus handler sees them.
    if (event is KeyDownEvent) {
      final target = trapKeys[event.logicalKey];
      if (target != null) {
        target.requestFocus();
        return true;
      }
    }
    if (event is KeyDownEvent) {
      final shift = HardwareKeyboard.instance.isShiftPressed ? 'shift+' : '';
      final name = event.logicalKey.keyLabel.isEmpty
          ? event.logicalKey.debugName ?? '?'
          : event.logicalKey.keyLabel;
      setState(() {
        _keys.add('$shift$name');
        if (_keys.length > 8) _keys.removeAt(0);
      });
    }
    return false;
  }

  Widget _line(String id, String text) => Semantics(
    identifier: id,
    label: text,
    container: true,
    child: ExcludeSemantics(
      child: Text(text, style: const TextStyle(fontSize: 11)),
    ),
  );

  @override
  Widget build(BuildContext context) {
    final open = openNow.value.toList()..sort();
    return Padding(
      padding: const EdgeInsets.all(4),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        mainAxisSize: MainAxisSize.min,
        children: [
          _line('open', 'Open: ${open.isEmpty ? '-' : open.join(' ')}'),
          _line('kf', 'KF: $_kf'),
          _line('log', 'Log: ${probeLog.value.join(', ')}'),
          _line('keys', 'Keys: ${_keys.join(' ')}'),
        ],
      ),
    );
  }
}

Widget button(String text, VoidCallback? onPressed, {FocusNode? focusNode}) =>
    Semantics(
      identifier: text,
      child: TextButton(
        focusNode: focusNode,
        onPressed: onPressed,
        child: Text(text),
      ),
    );

/// Runs [show], marks [id] open, and closed when the future completes.
Future<void> track(String id, Future<Object?> Function() show) async {
  opened(id);
  final result = await show();
  closed(id, result == null ? '' : ' ($result)');
}

Widget panel(String id, List<Widget> children) => Material(
  elevation: 8,
  child: Padding(
    padding: const EdgeInsets.all(16),
    child: Column(
      mainAxisSize: MainAxisSize.min,
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [Text('$id title'), ...children],
    ),
  ),
);

/// A policy that keeps Tab and Shift+Tab inside its group: next and
/// previous go round the group's own controls.
class _CyclePolicy extends WidgetOrderTraversalPolicy {
  @override
  bool next(FocusNode currentNode) => _move(currentNode, 1);

  @override
  bool previous(FocusNode currentNode) => _move(currentNode, -1);

  bool _move(FocusNode current, int by) {
    final group = FocusTraversalGroup.maybeOfNode(current);
    final scope = current.nearestScope;
    if (scope == null) return false;
    final nodes = sortDescendants(
      scope.traversalDescendants,
      current,
    ).where((n) => FocusTraversalGroup.maybeOfNode(n) == group).toList();
    final i = nodes.indexOf(current);
    if (nodes.isEmpty || i < 0) return false;
    nodes[(i + by) % nodes.length].requestFocus();
    return true;
  }
}

class TrapScreen extends StatefulWidget {
  const TrapScreen({super.key});

  @override
  State<TrapScreen> createState() => _TrapScreenState();
}

class _TrapScreenState extends State<TrapScreen> {
  final _t1a = FocusNode();
  final _t1b = FocusNode();
  final _t2a = FocusNode();
  final _t2b = FocusNode();
  final _after = FocusNode();
  final _t0 = FocusNode();
  final _t3a = FocusNode();
  final _t4a = FocusNode();
  final _t5a = FocusNode();
  final _t4 = FocusScopeNode(
    traversalEdgeBehavior: TraversalEdgeBehavior.closedLoop,
  );
  final _t5 = FocusScopeNode(
    traversalEdgeBehavior: TraversalEdgeBehavior.closedLoop,
  );

  @override
  void initState() {
    super.initState();
    trapKeys.addAll({
      LogicalKeyboardKey.f1: _t1a,
      LogicalKeyboardKey.f2: _t2a,
      LogicalKeyboardKey.f3: _t3a,
      LogicalKeyboardKey.f4: _t4a,
      LogicalKeyboardKey.f5: _t5a,
      LogicalKeyboardKey.f6: _t0,
    });
  }

  @override
  void dispose() {
    trapKeys.clear();
    for (final n in [
      _t1a,
      _t1b,
      _t2a,
      _t2b,
      _after,
      _t0,
      _t3a,
      _t4a,
      _t5a,
      _t4,
      _t5,
    ]) {
      n.dispose();
    }
    super.dispose();
  }

  /// t1 and t2: an onKeyEvent that takes Tab and Shift+Tab and moves
  /// between its two buttons. t2 also takes Escape and moves on.
  Widget _keyTrap(String id, FocusNode a, FocusNode b, {required bool escape}) {
    return Focus(
      canRequestFocus: false,
      skipTraversal: true,
      onKeyEvent: (node, event) {
        if (event is! KeyDownEvent && event is! KeyRepeatEvent) {
          return KeyEventResult.ignored;
        }
        if (event.logicalKey == LogicalKeyboardKey.tab) {
          (a.hasPrimaryFocus ? b : a).requestFocus();
          return KeyEventResult.handled;
        }
        if (escape && event.logicalKey == LogicalKeyboardKey.escape) {
          _after.requestFocus();
          log('$id escaped');
          return KeyEventResult.handled;
        }
        return KeyEventResult.ignored;
      },
      child: Row(
        children: [
          button('$id A', () => log('$id A'), focusNode: a),
          button('$id B', () => log('$id B'), focusNode: b),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('I08 traps')),
      body: ListView(
        children: [
          button('t0 before', () => log('t0'), focusNode: _t0),
          _keyTrap('t1', _t1a, _t1b, escape: false),
          _keyTrap('t2', _t2a, _t2b, escape: true),
          FocusTraversalGroup(
            policy: _CyclePolicy(),
            child: Row(
              children: [
                button('t3 A', () => log('t3 A'), focusNode: _t3a),
                button('t3 B', () => log('t3 B')),
              ],
            ),
          ),
          FocusScope(
            node: _t4,
            child: Row(
              children: [
                button('t4 A', () => log('t4 A'), focusNode: _t4a),
                button('t4 B', () => log('t4 B')),
              ],
            ),
          ),
          Actions(
            actions: {
              DismissIntent: CallbackAction<DismissIntent>(
                onInvoke: (_) {
                  _after.requestFocus();
                  log('t5 escaped');
                  return null;
                },
              ),
            },
            child: FocusScope(
              node: _t5,
              child: Row(
                children: [
                  button('t5 A', () => log('t5 A'), focusNode: _t5a),
                  button('t5 B', () => log('t5 B')),
                ],
              ),
            ),
          ),
          button('t9 after', () => log('t9'), focusNode: _after),
        ],
      ),
    );
  }
}
