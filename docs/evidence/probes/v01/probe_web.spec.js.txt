// V01 probe on the web (throwaway; kept as
// docs/evidence/probes/v01/probe_web.spec.js.txt, run with
// probe_web.config.js.txt against `flutter build web -t lib/probe.dart -o
// build/web_probe`). For every fixture of lib/probe.dart, light and dark:
// a screenshot at rest; for the fixtures that take input, a mouse hover, a
// mouse button held down, Tab focus, and Tab focus with the hover. Then for
// some fixtures, the accessibility tree from Chromium (role, name, states),
// to see what the candidate defects give assistive technology. Writes PNGs
// to V01_OUT for analyse.py and prints the tree.
import fs from 'node:fs';
import { test } from '@playwright/test';

const out = process.env.V01_OUT;
const ids = (process.env.V01_IDS ?? '').split(' ').filter(Boolean);
const inputs = new Set('m1 m2 m3 m4 m5 m32 m6 m12 m13 m14 m18 m19 m21 m22 m25 c1 c2 c3s c4s c5 c7 c9'.split(' '));
const trees = new Set('m13 m14 m18 m19 m21 m22 m23 m24 m25 m26 m27 c7 c8 c9 c10 c11 m10 m15 c5 m7 g2 g3'.split(' '));

test.describe.configure({ retries: 0 });
test.use({ viewport: { width: 400, height: 600 } });

async function open(page, route) {
  await page.goto('about:blank');
  await page.goto(`/#${route}`);
  await page.locator('flt-semantics').first().waitFor({ state: 'attached', timeout: 20000 });
  await page.waitForTimeout(1200);
}

/** Where the fixture's first control is: 40 by 20 logical pixels into the box. */
async function fixturePoint(page) {
  // The box is the first child after the title line; found from the canvas by
  // the title's semantics element, which sits 12 pixels above it.
  const title = page.locator('flt-semantics', { hasText: /^V01 / }).first();
  const r = await title.boundingBox();
  return { x: r.x + 3 + 16 + 40, y: r.y + r.height + 12 + 3 + 16 + 20 };
}

test('screenshots', async ({ page }) => {
  test.setTimeout(1_800_000);
  for (const id of ids) {
    for (const theme of ['light', 'dark']) {
      const route = `/f/${id}${theme === 'dark' ? '/dark' : ''}`;
      const tag = `web-${theme}-${id}`;
      await open(page, route);
      await page.mouse.move(399, 599);
      await page.screenshot({ path: `${out}/${tag}-rest.png` });
      if (!inputs.has(id)) continue;
      const p = await fixturePoint(page);
      await page.mouse.move(p.x, p.y);
      await page.waitForTimeout(800);
      await page.screenshot({ path: `${out}/${tag}-hover.png` });
      await page.mouse.down();
      await page.waitForTimeout(600);
      await page.screenshot({ path: `${out}/${tag}-pressed.png` });
      await page.mouse.up();
      await open(page, route);
      await page.mouse.move(399, 599);
      await page.keyboard.press('Tab');
      await page.waitForTimeout(800);
      await page.screenshot({ path: `${out}/${tag}-focused.png` });
      await page.mouse.move(p.x, p.y);
      await page.waitForTimeout(800);
      await page.screenshot({ path: `${out}/${tag}-focushover.png` });
    }
  }
});

test('trees', async ({ page }) => {
  test.setTimeout(600_000);
  const cdp = await page.context().newCDPSession(page);
  for (const id of ids.filter((i) => trees.has(i))) {
    await open(page, `/f/${id}`);
    await cdp.send('Accessibility.enable');
    const { nodes } = await cdp.send('Accessibility.getFullAXTree');
    const lines = [];
    for (const n of nodes) {
      if (n.ignored) continue;
      const role = n.role?.value;
      const name = n.name?.value ?? '';
      if (['generic', 'none', 'StaticText', 'InlineTextBox', 'RootWebArea', 'group'].includes(role) && !name) continue;
      const props = (n.properties ?? [])
        .filter((q) => ['checked', 'selected', 'pressed', 'invalid', 'disabled', 'focusable', 'expanded', 'required'].includes(q.name))
        .map((q) => `${q.name}=${q.value.value}`)
        .join(' ');
      lines.push(`  ${role} "${name}" ${props}`);
    }
    console.log(`V01TREE ${id}\n${lines.join('\n')}`);
    await cdp.send('Accessibility.disable');
  }
});
