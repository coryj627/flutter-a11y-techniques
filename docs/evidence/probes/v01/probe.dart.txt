// V01 probe (throwaway, not committed; kept as
// docs/evidence/probes/v01/probe.dart.txt): what colours Flutter's own
// widgets draw at their theme's defaults, light and dark, and what reaches
// assistive technology for the candidate defects.
//
// Route `/f/<id>` shows one fixture, `/f/<id>/dark` the same in the dark
// theme. The theme is the catalog app's: `ThemeData(colorSchemeSeed:
// Colors.indigo)`, light or dark, with no `CupertinoTheme` of its own, so
// the Cupertino widgets take the theme Material derives for them, as in the
// app. Fixtures whose id ends in `s` (c3s, c4s) put a default
// `CupertinoThemeData(brightness:)` round the fixture instead, whose primary
// colour is `systemBlue`.
//
// The fixture sits in a box, 16 logical pixels of the page round it, framed
// by a magenta (#FF00FF) line 3 logical pixels wide, so that the analyser
// can find the box in any screenshot. Nothing else is on the screen but a
// line at the top naming the fixture.
import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter/semantics.dart';

void main() {
  runApp(const ProbeApp());
  SemanticsBinding.instance.ensureSemantics();
}

const _seed = Colors.indigo;

ThemeData _theme(bool dark) => ThemeData(
  colorSchemeSeed: _seed,
  brightness: dark ? Brightness.dark : Brightness.light,
);

class ProbeApp extends StatelessWidget {
  const ProbeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: _theme(false),
      darkTheme: _theme(true),
      themeMode: ThemeMode.light,
      initialRoute: '/f/t1',
      onGenerateRoute: (settings) {
        final parts = (settings.name ?? '').split('/')
          ..removeWhere((p) => p.isEmpty);
        Widget screen = const Scaffold(body: Center(child: Text('probe')));
        if (parts.length >= 2 && parts[0] == 'f') {
          final dark = parts.length > 2 && parts[2] == 'dark';
          // As the app in dark mode: the dark theme, the Cupertino theme
          // derived from it, and the platform's brightness dark. A nested
          // `Theme` passes on the Cupertino theme above it rather than
          // deriving one from its own data (`material/theme.dart`,
          // `_inheritedCupertinoThemeData`), so round 1, which had the
          // nested `Theme` only, drew the Cupertino fixtures light in its
          // dark routes. Round 2 adds the derived Cupertino theme here.
          screen = Builder(
            builder: (context) => MediaQuery(
              data: MediaQuery.of(context).copyWith(
                platformBrightness: dark ? Brightness.dark : Brightness.light,
              ),
              child: CupertinoTheme(
                data: MaterialBasedCupertinoThemeData(
                  materialTheme: _theme(dark),
                ),
                child: Theme(
                  data: _theme(dark),
                  child: FixtureScreen(id: parts[1], dark: dark),
                ),
              ),
            ),
          );
        }
        return PageRouteBuilder<void>(
          settings: settings,
          pageBuilder: (_, _, _) => screen,
        );
      },
    );
  }
}

class FixtureScreen extends StatelessWidget {
  const FixtureScreen({super.key, required this.id, required this.dark});

  final String id;
  final bool dark;

  @override
  Widget build(BuildContext context) {
    final build = fixtures[id];
    Widget fixture = build == null
        ? Text('no fixture $id')
        : Builder(builder: (c) => build.$2(c));
    if (id.endsWith('s')) {
      fixture = CupertinoTheme(
        data: CupertinoThemeData(
          brightness: dark ? Brightness.dark : Brightness.light,
        ),
        child: fixture,
      );
    }
    return Scaffold(
      body: SafeArea(
        child: Padding(
          padding: const EdgeInsets.all(12),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text('V01 $id ${dark ? 'dark' : 'light'}'),
              const SizedBox(height: 12),
              DecoratedBox(
                decoration: BoxDecoration(
                  border: Border.all(color: const Color(0xFFFF00FF), width: 3),
                ),
                child: Padding(
                  padding: const EdgeInsets.all(16),
                  child: SizedBox(width: 280, child: fixture),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

Widget _col(List<Widget> c) => Column(
  crossAxisAlignment: CrossAxisAlignment.start,
  mainAxisSize: MainAxisSize.min,
  children: c,
);

void _noop() {}

const _gradient = LinearGradient(
  colors: [Color(0xFF1A237E), Color(0xFFC5CAE9)],
);

Widget _onGradient(Widget child) => DecoratedBox(
  decoration: const BoxDecoration(gradient: _gradient),
  child: SizedBox(height: 96, width: 280, child: child),
);
