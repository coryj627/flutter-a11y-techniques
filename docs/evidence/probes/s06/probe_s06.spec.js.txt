// S06 probe on the web (throwaway, not committed; kept as
// docs/evidence/probes/s06/probe_s06.spec.js.txt with
// probe_s06.config.js.txt). For each fixture of lib/probe.dart, its
// launcher opened by address: then the Open button clicked; then the app's
// Back button clicked; and, in a second test, Open clicked and the
// browser's own back and forward (page.goBack(), page.goForward()). At each
// step: document.title, the address, every title the document had at each
// change of the document since the last step (a MutationObserver), Chromium's
// accessibility tree root name, the semantics elements that carry a route
// (role, aria-describedby and what it points to), the headings, and the
// element with keyboard focus. p10: the second destination clicked; p13:
// the drawer opened and Escape pressed. A third test opens p01 and p05 with
// Tab and Enter.
import { test } from '@playwright/test';

const ids = Array.from({ length: 13 }, (_, i) => `p${String(i + 1).padStart(2, '0')}`);

function log(...a) {
  console.log('S06WEB', ...a.map((x) => String(x).replace(/\n/g, '\\n')));
}

const watched = new WeakSet();

async function fresh(page, id) {
  if (!watched.has(page)) {
    watched.add(page);
    await page.addInitScript(() => {
    window.__titles = [];
    let last = null;
    const seen = () => {
      if (document.title === last) return;
      last = document.title;
      window.__titles.push(`${(performance.now() / 1000).toFixed(2)}s "${document.title}"`);
    };
    new MutationObserver(seen).observe(document, { subtree: true, childList: true, characterData: true });
    });
  }
  await page.goto('about:blank');
  await page.goto(`/#/${id}`);
  await page.locator(`[flt-semantics-identifier="${id}-start"]`).waitFor();
  await page.waitForTimeout(1000);
}

async function state(context, page, id, step) {
  const info = await page.evaluate(() => {
    const titles = window.__titles.splice(0);
    const host = document.querySelector('flt-semantics-host');
    const routes = [...(host?.querySelectorAll('[aria-describedby], [role="dialog"], [role="alertdialog"]') ?? [])]
      .map((e) => {
        const d = e.getAttribute('aria-describedby');
        const target = d ? document.getElementById(d) : null;
        const tag = e.outerHTML.replace(/>[\s\S]*$/, '>').replace(/ style="[^"]*"/g, '');
        return `${tag} describedby-> ${target ? `"${target.getAttribute('aria-label') || target.textContent}"` : '(none)'}`;
      });
    const headings = [...(host?.querySelectorAll('h1,h2,h3,h4,h5,h6,[role="heading"]') ?? [])]
      .map((e) => `${e.tagName.toLowerCase()} "${e.textContent}"`);
    const a = document.activeElement;
    const active = a
      ? `${a.tagName.toLowerCase()} id=${a.getAttribute('flt-semantics-identifier') || ''} role=${a.getAttribute('role')} label="${a.getAttribute('aria-label') || (a.textContent || '').slice(0, 40)}"`
      : '(none)';
    return { title: document.title, href: location.href.replace(location.origin, ''), titles, routes, headings, active };
  });
  const cdp = await context.newCDPSession(page);
  const { nodes } = await cdp.send('Accessibility.getFullAXTree');
  const root = nodes.find((n) => n.role?.value === 'RootWebArea');
  const axHeadings = nodes.filter((n) => !n.ignored && n.role?.value === 'heading')
    .map((n) => `"${n.name?.value}" level=${(n.properties ?? []).find((p) => p.name === 'level')?.value?.value}`);
  const described = nodes.filter((n) => !n.ignored && n.description?.value)
    .map((n) => `role=${n.role?.value} name="${n.name?.value ?? ''}" description="${n.description?.value}"`);
  await cdp.detach();
  log(`${id} ${step}: title="${info.title}" href=${info.href}`);
  log(`${id} ${step}: titles since last step: ${info.titles.join(' | ') || '(none)'}`);
  log(`${id} ${step}: AX root name="${root?.name?.value}"`);
  log(`${id} ${step}: route elements: ${info.routes.join(' || ') || '(none)'}`);
  log(`${id} ${step}: AX nodes with a description: ${described.join(' || ') || '(none)'}`);
  log(`${id} ${step}: headings: ${info.headings.join(' | ') || '(none)'}; AX headings: ${axHeadings.join(' | ') || '(none)'}`);
  log(`${id} ${step}: keyboard focus: ${info.active}`);
}

async function clickOpen(page, id) {
  await page.locator(`[flt-semantics-identifier="${id}-open"]`).click();
  await page.locator(`[flt-semantics-identifier="${id}-end"]`).waitFor();
  await page.waitForTimeout(1200);
}

test('probe', async ({ context, page }) => {
  for (const id of ids) {
    await fresh(page, id);
    await state(context, page, id, 'launcher');
    await clickOpen(page, id);
    await state(context, page, id, 'opened');
    if (id === 'p10') {
      const candidates = await page.evaluate(() => [...document.querySelectorAll('flt-semantics')]
        .filter((e) => (e.getAttribute('aria-label') || e.textContent || '').includes('p10 Settings'))
        .map((e) => `${e.outerHTML.replace(/>[\s\S]*$/, '>').replace(/ style="[^"]*"/g, '')} text="${e.textContent}"`));
      log(`p10 elements with "p10 Settings": ${candidates.join(' || ')}`);
      // Round 2: the destination is role="tab" with its name in aria-label
      // and no text, so a filter by text never matched.
      await page.locator('flt-semantics[role="tab"][aria-label="p10 Settings"]').click({ timeout: 10_000 });
      await page.waitForTimeout(1200);
      await state(context, page, id, 'second destination');
    }
    if (id === 'p13') {
      await page.locator('[flt-semantics-identifier="p13-drawer"]').click();
      await page.waitForTimeout(1200);
      await state(context, page, id, 'drawer open');
      await page.keyboard.press('Escape');
      await page.waitForTimeout(1200);
      await state(context, page, id, 'drawer after Escape');
    }
    // p13 has no Back button: its app bar's leading button opens the drawer.
    if (id === 'p13') continue;
    await page.mouse.move(0, 0);
    // Round 3: a Cupertino back button has a second element named Back inside it.
    await page.getByRole('button', { name: 'Back' }).first().click();
    await page.waitForTimeout(1500);
    await state(context, page, id, 'app Back');
  }
});

test('browser back and forward', async ({ context, page }) => {
  for (const id of ids) {
    await fresh(page, id);
    await clickOpen(page, id);
    await state(context, page, id, 'opened (2nd test)');
    await page.goBack();
    await page.waitForTimeout(1500);
    await state(context, page, id, 'browser back');
    await page.goForward();
    await page.waitForTimeout(2000);
    await state(context, page, id, 'browser forward');
  }
});

test('as a Linux browser', async ({ context, page }) => {
  // navigator.platform, which Flutter's web engine reads for the platform
  // (ui_web browser_detection.dart), as on CI's Ubuntu runner.
  await page.addInitScript(() => {
    Object.defineProperty(Navigator.prototype, 'platform', { get: () => 'Linux x86_64' });
  });
  for (const id of ['p01', 'p05', 'p09', 'p13']) {
    await fresh(page, id);
    await clickOpen(page, id);
    await state(context, page, id, 'opened as Linux');
    if (id === 'p13') {
      await page.locator('[flt-semantics-identifier="p13-drawer"]').click();
      await page.waitForTimeout(1200);
      await state(context, page, id, 'drawer open as Linux');
    }
  }
});

test('keyboard open', async ({ context, page }) => {
  for (const id of ['p01', 'p05']) {
    await fresh(page, id);
    for (let i = 0; i < 6; i++) {
      await page.keyboard.press('Tab');
      const on = await page.evaluate(() => document.activeElement?.getAttribute('flt-semantics-identifier'));
      if (on === `${id}-open`) break;
    }
    await page.keyboard.press('Enter');
    await page.locator(`[flt-semantics-identifier="${id}-end"]`).waitFor();
    await page.waitForTimeout(1200);
    await state(context, page, id, 'opened by Enter');
  }
});
