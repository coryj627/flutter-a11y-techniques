// S11 probe (throwaway, not committed; kept as
// docs/evidence/probes/s11/ProbeS11UITests.swift.txt): for each fixture of
// lib/probe.dart, opened by URL: the XCUITest elements in the order XCUITest
// gives them, with their frames; for the fixtures with buttons, one to three
// Tabs then Space, from a fresh launch each, and which button the "After"
// line says was pressed (XCUITest cannot see Flutter's focus); and with
// VoiceOver on, a walk with moveForward from the top of the screen to the
// "After" line.
import XCTest

final class ProbeS11UITests: XCTestCase {
  let app = XCUIApplication()
  let ids = (1...20).map { String(format: "r%02d", $0) }
  let withButtons: Set<String> = ["r11", "r12", "r15", "r19", "r20"]

  func log(_ s: String) { print("S11PROBE \(s.replacingOccurrences(of: "\n", with: "\\n"))") }

  func after(_ id: String) -> XCUIElement { app.descendants(matching: .any)["\(id)-after"] }

  func open(_ id: String) {
    for _ in 1...3 {
      app.terminate()
      app.open(URL(string: "fluttera11y:///\(id)")!)
      if after(id).waitForExistence(timeout: 30) { sleep(1); return }
    }
    XCTFail("\(id) did not open")
  }

  func testTree() {
    for id in ids {
      open(id)
      var order: [String] = []
      for e in app.descendants(matching: .any).allElementsBoundByIndex {
        let t = e.elementType
        if t == .window || t == .application { continue }
        if !(e.label.contains("\(id) ") || e.label.hasPrefix("Before \(id)") || e.label.hasPrefix("After \(id)")) { continue }
        log("\(id) EL type=\(t.rawValue) label=\(e.label.debugDescription) frame=\(e.frame)")
        order.append(e.label)
      }
      log("\(id) element order: \(order.joined(separator: " / "))")
      if withButtons.contains(id) {
        for tabs in 1...3 {
          open(id)
          for _ in 0..<tabs {
            app.typeKey(XCUIKeyboardKey.tab.rawValue, modifierFlags: [])
            usleep(900_000)
          }
          app.typeKey(" ", modifierFlags: [])
          sleep(2)
          log("\(id) \(tabs) Tab(s) then Space: \(after(id).label)")
        }
      }
    }
  }

  func testVoiceOver() throws {
    #if compiler(>=6.4)
    guard #available(iOS 27.0, *) else { throw XCTSkip("needs iOS 27") }
    let vo = XCUIDevice.shared.voiceOverService
    func norm(_ s: String) -> String { s.split(whereSeparator: { $0.isWhitespace }).joined(separator: " ") }
    func say() -> String {
      do { return norm(try vo.moveForward().utterance) } catch { return "(error \((error as NSError).code))" }
    }
    for id in ids {
      open(id)
      try vo.enable()
      var walk: [String] = []
      for _ in 0..<24 {
        let s = say()
        walk.append(s)
        if s.hasPrefix("After \(id)") { break }
      }
      log("VO \(id) walk: \(walk.joined(separator: " / "))")
      try vo.disable()
    }
    #else
    throw XCTSkip("needs Xcode 27")
    #endif
  }
}
