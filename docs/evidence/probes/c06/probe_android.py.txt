"""C06 probe runner for Android (throwaway; kept as
docs/evidence/probes/c06/probe_android.py.txt). Run under the device lock:

  python3 tool/with_devices.py --label "probe C06 android" -- python3 <this> <repo> [tree|talkback|both]

Installs the probe build (lib/probe.dart, built beforehand with
`./gradlew app:assembleDebug app:assembleDebugAndroidTest -Ptarget=lib/probe.dart`)
and its test APK, then runs ProbeC06Test with `am instrument`. "keys" runs keys() (Tab, Enter, Space). "native" runs nativeTree() and nativeTalkBack() on the
native comparison screen (ProbeNativeActivity). "tree" runs
probe() without TalkBack and keeps the C06PROBE log lines; "talkback" turns
TalkBack on with the project's own helper (tool/talkback.py:
run_with_talkback, which restores the device), runs talkBackWalk(), and keeps
the speech per marker as parsed by tool/talkback.py's parse_speech.
"""
import os
import subprocess
import sys
import time

ROOT = sys.argv[1]
MODE = sys.argv[2] if len(sys.argv) > 2 else "both"
SUFFIX = sys.argv[3] if len(sys.argv) > 3 else ""
OUT = os.path.join(ROOT, "docs", "evidence", "probes", "c06")
sys.path.insert(0, os.path.join(ROOT, "tool"))
import talkback  # noqa: E402
from evidence_common import stop_android_tests  # noqa: E402

PKG = "dev.fluttera11y.flutter_a11y_techniques"
APKS = [
    os.path.join(ROOT, "build/app/outputs/apk/debug/app-debug.apk"),
    os.path.join(ROOT, "build/app/outputs/apk/androidTest/debug/app-debug-androidTest.apk"),
]


def instrument(method, log_path):
    talkback.adb("logcat", "-c")
    with open(log_path, "w") as log:
        reader = subprocess.Popen([talkback.ADB, "logcat"], stdout=log, stderr=subprocess.DEVNULL)
        try:
            run = subprocess.run(
                [talkback.ADB, "shell", "am", "instrument", "-w", "-e", "class",
                 f"{PKG}.ProbeC06Test#{method}", f"{PKG}.test/androidx.test.runner.AndroidJUnitRunner"],
                capture_output=True, text=True, timeout=30 * 60)
        finally:
            time.sleep(2)
            reader.terminate()
            reader.wait()
    print(run.stdout[-3000:])
    with open(log_path, errors="replace") as f:
        return run, f.read()


def keep(log, path):
    lines = [l for l in log.splitlines() if "C06PROBE" in l or "C06TB" in l]
    with open(path, "w") as f:
        f.write("\n".join(lines) + "\n")


def main():
    os.makedirs(OUT, exist_ok=True)
    scratch = os.path.dirname(os.path.abspath(__file__))
    if not stop_android_tests():
        sys.exit("emulator not clean")
    for apk in APKS:
        print(talkback.adb("install", "-r", "-t", apk)[-200:])
    print("device:", talkback.adb("shell", "getprop", "ro.build.version.release").strip(),
          talkback.adb("shell", "wm", "size").strip(), talkback.adb("shell", "wm", "density").strip())
    tree_method, walk_method = ("nativeTree", "nativeTalkBack") if MODE == "native" else ("probe", "talkBackWalk")
    if MODE == "keys":
        run, log = instrument("keys", os.path.join(scratch, "android-keys-logcat.txt"))
        keep(log, os.path.join(OUT, "android-keys.txt"))
        stop_android_tests()
    if MODE in ("tree", "both", "native"):
        run, log = instrument(tree_method, os.path.join(scratch, f"android-tree{SUFFIX}-logcat.txt"))
        keep(log, os.path.join(OUT, f"android-tree{SUFFIX}.txt"))
        stop_android_tests()
    if MODE in ("talkback", "both", "native"):
        talkback.exit_on_termination_signals()

        def action():
            return instrument(walk_method, os.path.join(scratch, f"android-talkback{SUFFIX}-logcat.txt"))

        run, log = talkback.run_with_talkback(action)
        stop_android_tests()
        spoken = talkback.parse_speech(log)
        out = ["| Marker | TalkBack spoke |", "| --- | --- |"]
        for ident, fragments in spoken.items():
            out.append(f"| `{ident}` | {' / '.join(fragments) or '(nothing)'} |")
        with open(os.path.join(OUT, f"android-talkback{SUFFIX}.md"), "w") as f:
            f.write("\n".join(out) + "\n")
        print("\n".join(out))
    talkback.adb("uninstall", f"{PKG}.test")


if __name__ == "__main__":
    main()
