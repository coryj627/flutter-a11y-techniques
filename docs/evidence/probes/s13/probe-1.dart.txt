import 'package:flutter/material.dart';

void main() {

  SemanticsBinding.instance.ensureSemantics();
}

class ProbeScreen extends StatefulWidget {
  const ProbeScreen({super.key});

  @override
  State<ProbeScreen> createState() => _ProbeScreenState();
}

    child: Align(alignment: Alignment.centerLeft, child: Text(text)),

        child: SingleChildScrollView(

            spacing: 4,
            runSpacing: 2,
            children: [
              Semantics(container: true, button: true, onTap: () => _did('01'), child: _box('n01 button')),
              Semantics(container: true, link: true, onTap: () => _did('02'), child: _box('n02 link')),
              Semantics(
                container: true,
                link: true,
                linkUrl: Uri.parse('https://example.com/'),
                onTap: () => _did('03'),
                child: _box('n03 link url'),
              ),
              Semantics(container: true, image: true, child: _box('n04 image')),
              Semantics(container: true, selected: true, onTap: () => _did('05'), child: _box('n05 selected')),

              ),
              Semantics(container: true, keyboardKey: true, onTap: () => _did('10'), child: _box('n10 key')),
              Semantics(
                container: true,
                keyboardKey: true,
                button: true,
                onTap: () => _did('11'),
                child: _box('n11 key button'),
              ),
              Semantics(
                container: true,
                slider: true,
                value: '5',
                increasedValue: '6',
                decreasedValue: '4',
                onIncrease: () => _did('12+'),
                onDecrease: () => _did('12-'),
