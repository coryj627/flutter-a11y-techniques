// Throwaway, C02 review round 3 (kept as
// docs/evidence/probes/c02/icon_foreground_test.dart.txt): the colour a
// CupertinoButton's text and icon are painted in when its foregroundColor
// is a CupertinoDynamicColor that is not resolved, under a MaterialApp and
// under a CupertinoApp, light and dark; and the same for good-cupertino as
// it is now, once on, in the app.
import 'dart:math' as math;

import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter_a11y_techniques/techniques/c02_toggle_button/c02_toggle_button.dart';
import 'package:flutter_test/flutter_test.dart';

import 'support/examples.dart';

double _lum(Color c) {
  double ch(double v) =>
      v <= 0.03928 ? v / 12.92 : math.pow((v + 0.055) / 1.055, 2.4).toDouble();
  return 0.2126 * ch(c.r) + 0.7152 * ch(c.g) + 0.0722 * ch(c.b);
}

double _ratio(Color a, Color b) {
  final la = _lum(a), lb = _lum(b);
  return (math.max(la, lb) + 0.05) / (math.min(la, lb) + 0.05);
}

String _rgb(Color c) =>
    '(${c.r.toStringAsFixed(3)}, ${c.g.toStringAsFixed(3)}, ${c.b.toStringAsFixed(3)})';

/// The colours actually used: the text's from its RichText, the icon's as
/// Icon reads it (IconTheme.of at the icon), the fill as the button
/// resolves it.
void _report(WidgetTester tester, String what) {
  final buttonFinder = find.byType(CupertinoButton).first;
  final button = tester.widget<CupertinoButton>(buttonFinder);
  final fill = CupertinoDynamicColor.resolve(button.color!, tester.element(buttonFinder));
  final text = tester
      .widget<RichText>(find.descendant(of: buttonFinder, matching: find.byType(RichText)).last)
      .text
      .style!
      .color!;
  final iconElement = tester.element(find.descendant(of: buttonFinder, matching: find.byType(Icon)));
  final icon = (tester.widget<Icon>(find.descendant(of: buttonFinder, matching: find.byType(Icon))).color) ??
      IconTheme.of(iconElement).color!;
  final iconTheme = IconTheme.of(iconElement).runtimeType;
  debugPrint('C02ICON $what | fill=${_rgb(fill)} text=${_rgb(text)} ratio ${_ratio(text, fill).toStringAsFixed(2)} '
      '| icon=${_rgb(icon)} ratio ${_ratio(icon, fill).toStringAsFixed(2)} | icon theme at the icon: $iconTheme');
}

Widget _button() => CupertinoButton(
  color: CupertinoColors.label,
  foregroundColor: CupertinoColors.systemBackground,
  onPressed: () {},
  child: const Row(
    mainAxisSize: MainAxisSize.min,
    children: [Icon(CupertinoIcons.speaker_slash_fill), Text('Mute')],
  ),
);

void main() {
  for (final brightness in Brightness.values) {
    testWidgets('C02ICON unresolved foregroundColor, MaterialApp, ${brightness.name}', (tester) async {
      tester.platformDispatcher.platformBrightnessTestValue = brightness;
      addTearDown(tester.platformDispatcher.clearAllTestValues);
      await tester.pumpWidget(MaterialApp(
        theme: ThemeData(brightness: Brightness.light),
        darkTheme: ThemeData(brightness: Brightness.dark),
        home: Scaffold(body: Center(child: _button())),
      ));
      _report(tester, 'unresolved foregroundColor, MaterialApp, ${brightness.name}');
    });

    testWidgets('C02ICON unresolved foregroundColor, CupertinoApp, ${brightness.name}', (tester) async {
      tester.platformDispatcher.platformBrightnessTestValue = brightness;
      addTearDown(tester.platformDispatcher.clearAllTestValues);
      await tester.pumpWidget(CupertinoApp(home: CupertinoPageScaffold(child: Center(child: _button()))));
      _report(tester, 'unresolved foregroundColor, CupertinoApp, ${brightness.name}');
    });

    testWidgets('C02ICON good-cupertino as it is now, on, in the app, ${brightness.name}', (tester) async {
      await exampleOpener(c02ToggleButton)(tester, 'good-cupertino', brightness: brightness);
      await tester.tap(byId('c02-good-cupertino-mute'));
      await tester.pumpAndSettle();
      _report(tester, 'good-cupertino now, on, app, ${brightness.name}');
    });
  }
}
