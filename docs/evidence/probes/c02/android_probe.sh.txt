#!/bin/sh
# C02 probe on Android, under tool/with_devices.py. Run from the worktree
# root, after `fvm flutter build apk --debug -t lib/probe.dart` (with the
# throwaway ProbeNativeActivity declared in the debug manifest). Installs the
# probe, dumps the screen's node tree, taps "Flip all", and dumps it again,
# into android_before.xml and android_after.xml; then opens the native
# screen and dumps it before and after a click on n05, into
# android_native_before.xml and android_native_after.xml.
# Changes no device setting.
PKG=dev.fluttera11y.flutter_a11y_techniques
OUT=docs/evidence/probes/c02
echo "== device: Android $(adb shell getprop ro.build.version.release | tr -d '\r'), $(adb shell wm size | tr -d '\r'), $(adb shell wm density | tr -d '\r')"
adb install -r build/app/outputs/flutter-apk/app-debug.apk | tail -1
centre() {
  # The centre of the first node whose text or content-desc is $2, in file $1.
  B=$(tr '>' '\n' < "$1" | grep -E "(text|content-desc)=\"$2\"" | head -1 | grep -oE 'bounds="\[[0-9]+,[0-9]+\]\[[0-9]+,[0-9]+\]"' | grep -oE '[0-9]+' | paste -sd ' ' -)
  set -- $B
  echo "$(( ($1 + $3) / 2 )) $(( ($2 + $4) / 2 ))"
}
adb shell am force-stop $PKG
adb shell monkey -p $PKG 1 >/dev/null 2>&1
sleep 8
adb exec-out uiautomator dump /dev/tty > $OUT/android_before.xml
AT=$(centre $OUT/android_before.xml "Flip all")
echo "== Flip all at $AT"
adb shell input tap $AT
sleep 3
adb exec-out uiautomator dump /dev/tty > $OUT/android_after.xml
adb shell am force-stop $PKG
adb shell am start -n $PKG/.ProbeNativeActivity >/dev/null
sleep 4
adb exec-out uiautomator dump /dev/tty > $OUT/android_native_before.xml
AT=$(centre $OUT/android_native_before.xml "n05 Bold")
echo "== n05 at $AT"
adb shell input tap $AT
sleep 2
adb exec-out uiautomator dump /dev/tty > $OUT/android_native_after.xml
adb shell am force-stop $PKG
echo "== done"
